33// Cells within this of 0 or 1 count as pure. It must stay well above the noise in F: otherwise a
34// cell flips between mixed and pure from one neighbour to the next, which breaks the symmetry of
35// the flow. The noise is about 1e-11 in double precision.
36#ifdef AMREX_USE_FLOAT
37static constexpr amrex::Real pure_tol = 1.0e-4_rt;
38#else
39static constexpr amrex::Real pure_tol = 1.0e-8_rt;
40#endif

42/** \brief Whether a cell with volume fraction f holds the interface. */
43AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool is_mixed(const amrex::Real f) noexcept {
44 return f > pure_tol && f < 1.0_rt - pure_tol;
45}

47/** \brief a, b and c in ascending order, as lo <= mid <= hi. */
48AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void sort3(const amrex::Real a,
49 const amrex::Real b,
50 const amrex::Real c,
51 amrex::Real& lo,
52 amrex::Real& mid,
53 amrex::Real& hi) noexcept {
54 lo = amrex::min(a, b);
55 hi = amrex::max(a, b);
56 mid = c;
57 if (mid < lo) {
58 amrex::Swap(lo, mid);
59 } else if (mid > hi) {
60 amrex::Swap(mid, hi);
61 }
62}

64/**
65 * \brief The root of the cubic of volume_intercept in its two upper volume ranges, in the
66 * trigonometric form of Scardovelli & Zaleski, without the constant offset of the range.
67 */
68AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real trig_root(const amrex::Real p,
69 const amrex::Real q) noexcept {
70 const amrex::Real p12 = std::sqrt(amrex::max(0.0_rt, p));
71 const amrex::Real pp = p * p12;
72 const amrex::Real teta =
73 std::acos((pp > 0.0_rt) ? amrex::Clamp(q / pp, -1.0_rt, 1.0_rt) : 1.0_rt) / 3.0_rt;
74 const amrex::Real cs = std::cos(teta);
75 return p12 * (std::sqrt(3.0_rt * (1.0_rt - cs * cs)) - cs);
76}

78// Cells searched along a wall, on either side of a wall ghost cell, for the plane it extends (see
79// wall_contact_planes). With a >= b the slopes of the interface relative to the wall:
80// - a cell R layers inside the wall reads no ghost cells (R = 4 for myc_height), and the interface
81// reaches one within R/a + 1 cells: found for a >= R / (wall_nsearch - 1);
82// - nearer the wall, the MYC plane is exact while its 3-cell columns hold the interface,
83// 1/2 + 2a + b <= 3/2: for 2a + b <= 1.
84// Together they cover every angle for wall_nsearch >= 2R + 1 = 9 (asserted after HeightFunctionMYC;
85// with a tilt the estimate needs about 9.5, but 9 is exact over random planes in Tests/PLIC). Each
86// cell more costs FillWallGhosts about 10% on the CPU and 5% on the GPU.
87static constexpr int wall_nsearch = 9;

89/**
90 * \brief Analytic plane constant alpha such that the region
91 * m1*x1 + m2*x2 + m3*x3 <= alpha of the unit cube has volume volF.
92 * Requires m1,m2,m3 >= 0 and m1+m2+m3 = 1. Any one (2D) or two
93 * (grid-aligned) coefficients may be zero.
94 */
95AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real volume_intercept(const amrex::Real b1,
96 const amrex::Real b2,
97 const amrex::Real b3,
98 const amrex::Real volF) {
99 constexpr amrex::Real tiny = std::numeric_limits<amrex::Real>::epsilon();
100
101 amrex::Real m1, m2, m3;
102 sort3(b1, b2, b3, m1, m2, m3);
103
104 // Volume at the breakpoints V1 < V2 < V3
105 const amrex::Real m12 = m1 + m2;
106 const amrex::Real pr = 6.0_rt * m1 * m2 * m3;
107 // Without a corner region (m1 = 0, a 2D or grid-aligned plane), V1 = 0
108 const amrex::Real V1 = (pr > 0.0_rt) ? m1 * m1 * m1 / pr : 0.0_rt;
109 const amrex::Real V2 = V1 + 0.5_rt * (m2 - m1) / (m3 + tiny);
110 amrex::Real V3 = 0.0_rt;
111 if (m3 < m12) {
112 // cut_volume at alpha = m3, without cancellation (m1 > 0)
113 const amrex::Real t2 = m3 - m2;
114 V3 = (3.0_rt * m3 * (m3 - m1) + m1 * m1 - t2 * t2 * (t2 / m1)) / (6.0_rt * m2 * m3);
115 } else {
116 V3 = 0.5_rt * m12 / (m3 + tiny);
117 }
118
119 // Use symmetry to work with ch <= 1/2
120 const amrex::Real ch = amrex::min(volF, 1.0_rt - volF);
121
122 amrex::Real alpha = 0.0_rt;
123 if (ch < V1) {
124 alpha = std::cbrt(pr * ch);
125 } else if (ch < V2) {
126 alpha = 0.5_rt * (m1 + std::sqrt(m1 * m1 + 8.0_rt * m2 * m3 * (ch - V1)));
127 } else if (ch < V3) {
128 const amrex::Real p = 2.0_rt * m1 * m2;
129 const amrex::Real q = 1.5_rt * m1 * m2 * (m12 - 2.0_rt * m3 * ch);
130 alpha = trig_root(p, q) + m12;
131 } else if (m12 < m3) {
132 alpha = m3 * ch + 0.5_rt * m12;
133 } else {
134 const amrex::Real p = m1 * (m2 + m3) + m2 * m3 - 0.25_rt;
135 const amrex::Real q = 1.5_rt * m1 * m2 * m3 * (0.5_rt - ch);
136 alpha = trig_root(p, q) + 0.5_rt;
137 }
138
139 if (volF > 0.5_rt) {
140 alpha = 1.0_rt - alpha;
141 }
142 return alpha;
143}

145/**
146 * \brief Liquid volume fraction of the sub-box [r0, r0+dr0] x [0,1] x [0,1]
147 * of the unit cube cut by m1*x1 + m2*x2 + m3*x3 <= alpha. The
148 * result is normalized by the sub-box volume dr0. Coefficients may
149 * have any sign; m3 = 0 gives the 2D result.
150 */
151AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real cut_volume(const amrex::Real m1,
152 const amrex::Real m2,
153 const amrex::Real m3,
154 const amrex::Real alpha,
155 const amrex::Real r0,
156 const amrex::Real dr0) {
157 constexpr amrex::Real tiny = std::numeric_limits<amrex::Real>::epsilon();
158
159 // Move origin to r0, then reflect so that all coefficients are positive
160 amrex::Real al = alpha - m1 * r0;
161 al += amrex::max(0.0_rt, -m1 * dr0) + amrex::max(0.0_rt, -m2) + amrex::max(0.0_rt, -m3);
162
163 // Normalize to the unit cube: n1*y1 + n2*y2 + n3*y3 = al, sum(n) = 1
164 const amrex::Real tmp = amrex::max(tiny, std::abs(m1) * dr0 + std::abs(m2) + std::abs(m3));
165 const amrex::Real n1 = std::abs(m1) / tmp;
166 const amrex::Real n2 = std::abs(m2) / tmp;
167 const amrex::Real n3 = std::abs(m3) / tmp;
168 al = amrex::max(0.0_rt, amrex::min(1.0_rt, al / tmp));
169 const amrex::Real a = amrex::min(al, 1.0_rt - al);
170
171 amrex::Real b1, b2, b3;
172 sort3(n1 * dr0, n2, n3, b1, b2, b3);
173 // Components below eps^2 change the volume by less than that, and would underflow
174 constexpr amrex::Real negligible = tiny * tiny;
175 if (b2 < negligible) {
176 b2 = 0.0_rt;
177 }
178 if (b1 < negligible) {
179 b1 = 0.0_rt;
180 }
181 const amrex::Real b12 = b1 + b2;
182
183 // Scardovelli & Zaleski (JCP 2000), as arranged by Kawano (C&F 2016). Between b2 and b12
184 // (where b1 > 0), the cubic is divided through by b1 to avoid cancellation for small b1.
185 amrex::Real v = 0.0_rt;
186 if (a > 0.0_rt) {
187 if (a < b1) {
188 v = a * a * a / (6.0_rt * b1 * b2 * b3);
189 } else if (a < b2) {
190 v = a * (a - b1) / (2.0_rt * b2 * b3) + b1 * b1 / (6.0_rt * b2 * b3);
191 } else if (a < amrex::min(b12, b3)) {
192 const amrex::Real t2 = a - b2;
193 v = (3.0_rt * a * (a - b1) + b1 * b1 - t2 * t2 * (t2 / b1)) / (6.0_rt * b2 * b3);
194 } else if (b3 < b12) {
195 const amrex::Real t2 = a - b2;
196 const amrex::Real t3 = a - b3;
197 v = (3.0_rt * a * (a - b1) + b1 * b1 - t2 * t2 * (t2 / b1) - t3 * t3 * (t3 / b1)) /
198 (6.0_rt * b2 * b3);
199 } else {
200 v = (a - 0.5_rt * b12) / b3;
201 }
202 }
203
204 return (al <= 0.5_rt) ? v : 1.0_rt - v;
205}

207/**
208 * \brief Liquid volume fraction of cell (i,j,k) for an implicit surface
209 * phi(x) > 0 inside the liquid, by midpoint sampling on nsub^dim
210 * sub-cells. Intended for problem initialization.
211 */
212template <typename Phi>
213AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
214cell_fraction(const int i,
215 const int j,
216 const int k,
217 amrex::GeometryData const& geomdata,
218 Phi const& phi,
219 const int nsub = 8) {
220 const amrex::Real* prob_lo = geomdata.ProbLo();
221 const amrex::Real* dx = geomdata.CellSize();
222 AMREX_ASSERT(nsub > 0);
223 const amrex::Real h = 1.0_rt / static_cast<amrex::Real>(nsub);
224 int inside = 0;
225#if AMREX_SPACEDIM == 3
226 for (int kk = 0; kk < nsub; ++kk) {
227 const amrex::Real z =
228 prob_lo[2] +
229 (static_cast<amrex::Real>(k) + (static_cast<amrex::Real>(kk) + 0.5_rt) * h) * dx[2];
230#else
231 amrex::ignore_unused(k);
232 {
233 const amrex::Real z = 0.0_rt;
234#endif
235 for (int jj = 0; jj < nsub; ++jj) {
236 const amrex::Real y =
237 prob_lo[1] +
238 (static_cast<amrex::Real>(j) + (static_cast<amrex::Real>(jj) + 0.5_rt) * h) * dx[1];
239 for (int ii = 0; ii < nsub; ++ii) {
240 const amrex::Real x = prob_lo[0] + (static_cast<amrex::Real>(i) +
241 (static_cast<amrex::Real>(ii) + 0.5_rt) * h) *
242 dx[0];
243 if (phi(x, y, z) > 0.0_rt) {
244 ++inside;
245 }
246 }
247 }
248 }
249 return static_cast<amrex::Real>(inside) /
250 static_cast<amrex::Real>(AMREX_D_TERM(nsub, *nsub, *nsub));
251}

253/**
254 * \brief Exact liquid volume fraction of cell (i,j,k) for the half space
255 * n.x <= d (physical coordinates, n need not be normalized).
256 * Intended for problem initialization.
257 */
258AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
259plane_fraction(const int i,
260 const int j,
261 const int k,
262 amrex::GeometryData const& geomdata,
263 const amrex::Real n[AMREX_SPACEDIM],
264 const amrex::Real d) {
265 const amrex::Real* prob_lo = geomdata.ProbLo();
266 const amrex::Real* dx = geomdata.CellSize();
267 const int idx[3] = {i, j, k};
268 amrex::Real m[3] = {0.0_rt, 0.0_rt, 0.0_rt};
269 amrex::Real alpha = d;
270 amrex::Real norm = 0.0_rt;
271 for (int a = 0; a < AMREX_SPACEDIM; ++a) {
272 m[a] = n[a] * dx[a];
273 alpha -= n[a] * (prob_lo[a] + static_cast<amrex::Real>(idx[a]) * dx[a]);
274 norm += std::abs(m[a]);
275 }
276 if (norm <= std::numeric_limits<amrex::Real>::min()) {
277 return (alpha >= 0.0_rt) ? 1.0_rt : 0.0_rt;
278 }
279 return amrex::Clamp(
280 cut_volume(m[0] / norm, m[1] / norm, m[2] / norm, alpha / norm, 0.0_rt, 1.0_rt), 0.0_rt,
281 1.0_rt);
282}

284/**
285 * \brief Liquid volume fraction of cell c under the plane (m, alpha) given in the unit-cube
286 * coordinates of cell ref.
287 */
288AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real plane_volume_in(const amrex::IntVect& c,
289 const amrex::IntVect& ref,
290 const amrex::Real* m,
291 const amrex::Real alpha) {
292 amrex::Real alpha_c = alpha;
293 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
294 alpha_c -= m[d] * static_cast<amrex::Real>(c[d] - ref[d]);
295 }
296 return cut_volume(m[0], m[1], m[2], alpha_c, 0.0_rt, 1.0_rt);
297}

299/**
300 * \brief Volume fraction of the ghost cell g beyond the wall, from the plane (m, alpha) of cell ref
301 * (see wall_contact_planes). The ghost takes the change of the plane's volume from the wall
302 * cell anchor: the same as the plane's volume at g for a planar interface, and the value of
303 * the anchor where the plane does not vary across the wall, whichever cell the plane comes
304 * from, so that a flow uniform normal to the wall stays uniform.
305 *
306 * \param F_w volume fraction of the anchor
307 * \param V_w plane_volume_in(anchor, ref, m, alpha), the same for every ghost cell of the anchor
308 */
309AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real wall_ghost_fraction(const amrex::IntVect& g,
310 const amrex::IntVect& ref,
311 const amrex::Real* m,
312 const amrex::Real alpha,
313 const amrex::Real F_w,
314 const amrex::Real V_w) {
315 const amrex::Real V_g = plane_volume_in(g, ref, m, alpha);
316 return amrex::Clamp(F_w + V_g - V_w, 0.0_rt, 1.0_rt);
317}

319/**
320 * \brief Liquid fraction of the half segment from the center of cell (i,j,k) to the center of its
321 * face in direction dir on side sgn (+1 high, -1 low), from the plane of the cell.
322 *
323 * \param plane the PLIC plane of each mixed cell (m_x, m_y, m_z, alpha)
324 */
325AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
326half_segment_fraction(const int i,
327 const int j,
328 const int k,
329 const int dir,
330 const int sgn,
331 amrex::Array4<amrex::Real const> const& F,
332 amrex::Array4<amrex::Real const> const& plane) {
333 const amrex::Real Fc = F(i, j, k);
334 if (Fc <= pure_tol) {
335 return 0.0_rt;
336 }
337 if (Fc >= 1.0_rt - pure_tol) {
338 return 1.0_rt;
339 }
340 const amrex::Real m[3] = {plane(i, j, k, 0), plane(i, j, k, 1), plane(i, j, k, 2)};
341 const amrex::Real alpha = plane(i, j, k, 3);
342 // psi = m.x - alpha < 0 in the liquid, linear along the half segment
343 const amrex::Real psi_c = 0.5_rt * (m[0] + m[1] + m[2]) - alpha;
344 const amrex::Real psi_f = psi_c + 0.5_rt * static_cast<amrex::Real>(sgn) * m[dir];
345 if (psi_c <= 0.0_rt && psi_f <= 0.0_rt) {
346 return 1.0_rt;
347 }
348 if (psi_c > 0.0_rt && psi_f > 0.0_rt) {
349 return 0.0_rt;
350 }
351 const amrex::Real t = psi_c / (psi_c - psi_f);
352 return (psi_c <= 0.0_rt) ? t : 1.0_rt - t;
353}

355/**
356 * \brief Liquid fraction of the segment joining the centers of the cells on either side of the low
357 * face of (i,j,k) in direction dir. Exact for a planar interface, which makes
358 * rho_face * g a discrete gradient (hydrostatic balance) when
359 * rho_face = rho_g + (rho_l - rho_g) * this.
360 */
361AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
362face_segment_fraction(const int i,
363 const int j,
364 const int k,
365 const int dir,
366 amrex::Array4<amrex::Real const> const& F,
367 amrex::Array4<amrex::Real const> const& plane) {
368 const amrex::IntVect iv(AMREX_D_DECL(i, j, k));
369 const auto m = (iv - amrex::IntVect::TheDimensionVector(dir)).dim3();
370 return 0.5_rt * (half_segment_fraction(m.x, m.y, m.z, dir, 1, F, plane) +
371 half_segment_fraction(i, j, k, dir, -1, F, plane));
372}
