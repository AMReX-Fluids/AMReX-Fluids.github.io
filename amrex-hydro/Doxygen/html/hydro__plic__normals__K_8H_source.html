30// Half-length of the height-function columns (7 cells)
31static constexpr int hf_half = 3;
32// A height column ends in cells within this of 0 or 1
33#ifdef AMREX_USE_FLOAT
34static constexpr amrex::Real hf_pure_tol = 1.0e-3_rt;
35#else
36static constexpr amrex::Real hf_pure_tol = 1.0e-6_rt;
37#endif
38// Relative difference below which two values tie, to round-off (see tied)
39static constexpr amrex::Real tie_tol = 1000.0_rt * std::numeric_limits<amrex::Real>::epsilon();

41//! Whether a ties with the largest of its kind amax (to round-off)
42AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool tied(const amrex::Real a, const amrex::Real amax) noexcept {
43 return a >= (1.0_rt - tie_tol) * amax;
44}

46/**
47 * \brief The MYC normal from its centered candidates m[c] (height direction c, c < n, height
48 * component +-1) and the Youngs normal m[n] (at any scale): the centered candidate whose
49 * height component dominates after L1 normalization, or the mean of those that tie, so that
50 * no direction is preferred, unless Youngs' largest component is smaller (an interface not
51 * aligned with a grid plane).
52 */
53template <int n>
54AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void myc_choose(amrex::Real (&m)[n + 1][3],
55 amrex::Real& mx,
56 amrex::Real& my,
57 amrex::Real& mz) {
58 // The L1 norm of each candidate, and the components compared after normalization: only the
59 // candidates kept are normalized in full
60 amrex::Real t[n + 1];
61 amrex::Real hc[n];
62 amrex::Real t0 = 0.0_rt;
63 for (int c = 0; c <= n; ++c) {
64 t[c] = std::abs(m[c][0]) + std::abs(m[c][1]) + std::abs(m[c][2]);
65 if (c < n) {
66 hc[c] = std::abs(m[c][c]) / t[c];
67 t0 = amrex::max(t0, hc[c]);
68 }
69 }
70 const amrex::Real ty =
71 (t[n] > 0.0_rt)
72 ? amrex::max(std::abs(m[n][0]), std::abs(m[n][1]), std::abs(m[n][2])) / t[n]
73 : 0.0_rt;
74 amrex::Real mc[3] = {0.0_rt, 0.0_rt, 0.0_rt};
75 if (ty > 0.0_rt && t0 > ty) {
76 for (int d = 0; d < 3; ++d) {
77 mc[d] = m[n][d] / t[n];
78 }
79 } else {
80 for (int c = 0; c < n; ++c) {
81 if (tied(hc[c], t0)) {
82 for (int d = 0; d < 3; ++d) {
83 mc[d] += m[c][d] / t[c];
84 }
85 }
86 }
87 }
88 // Tied candidates may cancel: a null normal, which fit_plane replaces
89 const amrex::Real tc = std::abs(mc[0]) + std::abs(mc[1]) + std::abs(mc[2]);
90 const amrex::Real scale = (tc > 0.0_rt) ? 1.0_rt / tc : 0.0_rt;
91 mx = mc[0] * scale;
92 my = mc[1] * scale;
93 mz = mc[2] * scale;
94}

146/**
147 * \brief Mixed Youngs-Centered normal (3D). Returns an L1-normalized
148 * normal pointing out of the liquid.
149 */
150AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void myc_normal(const int i,
151 const int j,
152 const int k,
153 amrex::Array4<amrex::Real const> const& F,
154 amrex::Real& mx,
155 amrex::Real& my,
156 amrex::Real& mz) {
157 // m[c][d]: candidate c (0-2 centered height-function schemes along x, y,
158 // z, 3 Youngs), component d
159 amrex::Real m[4][3];
160
161 // Centered scheme with x as the height direction:
162 // sgn(mx) X = my Y + mz Z + alpha
163 amrex::Real m1 = F(i - 1, j, k - 1) + F(i - 1, j, k + 1) + F(i - 1, j - 1, k) +
164 F(i - 1, j + 1, k) + F(i - 1, j, k);
165 amrex::Real m2 = F(i + 1, j, k - 1) + F(i + 1, j, k + 1) + F(i + 1, j - 1, k) +
166 F(i + 1, j + 1, k) + F(i + 1, j, k);
167 m[0][0] = (m1 > m2) ? 1.0_rt : -1.0_rt;
168 m1 = F(i - 1, j - 1, k) + F(i + 1, j - 1, k) + F(i, j - 1, k);
169 m2 = F(i - 1, j + 1, k) + F(i + 1, j + 1, k) + F(i, j + 1, k);
170 m[0][1] = 0.5_rt * (m1 - m2);
171 m1 = F(i - 1, j, k - 1) + F(i + 1, j, k - 1) + F(i, j, k - 1);
172 m2 = F(i - 1, j, k + 1) + F(i + 1, j, k + 1) + F(i, j, k + 1);
173 m[0][2] = 0.5_rt * (m1 - m2);
174
175 // y as the height direction
176 m1 = F(i - 1, j - 1, k) + F(i - 1, j + 1, k) + F(i - 1, j, k);
177 m2 = F(i + 1, j - 1, k) + F(i + 1, j + 1, k) + F(i + 1, j, k);
178 m[1][0] = 0.5_rt * (m1 - m2);
179 m1 = F(i, j - 1, k - 1) + F(i, j - 1, k + 1) + F(i + 1, j - 1, k) + F(i - 1, j - 1, k) +
180 F(i, j - 1, k);
181 m2 = F(i, j + 1, k - 1) + F(i, j + 1, k + 1) + F(i + 1, j + 1, k) + F(i - 1, j + 1, k) +
182 F(i, j + 1, k);
183 m[1][1] = (m1 > m2) ? 1.0_rt : -1.0_rt;
184 m1 = F(i, j - 1, k - 1) + F(i, j, k - 1) + F(i, j + 1, k - 1);
185 m2 = F(i, j - 1, k + 1) + F(i, j, k + 1) + F(i, j + 1, k + 1);
186 m[1][2] = 0.5_rt * (m1 - m2);
187
188 // z as the height direction
189 m1 = F(i - 1, j, k - 1) + F(i - 1, j, k + 1) + F(i - 1, j, k);
190 m2 = F(i + 1, j, k - 1) + F(i + 1, j, k + 1) + F(i + 1, j, k);
191 m[2][0] = 0.5_rt * (m1 - m2);
192 m1 = F(i, j - 1, k - 1) + F(i, j - 1, k + 1) + F(i, j - 1, k);
193 m2 = F(i, j + 1, k - 1) + F(i, j + 1, k + 1) + F(i, j + 1, k);
194 m[2][1] = 0.5_rt * (m1 - m2);
195 m1 = F(i - 1, j, k - 1) + F(i + 1, j, k - 1) + F(i, j - 1, k - 1) + F(i, j + 1, k - 1) +
196 F(i, j, k - 1);
197 m2 = F(i - 1, j, k + 1) + F(i + 1, j, k + 1) + F(i, j - 1, k + 1) + F(i, j + 1, k + 1) +
198 F(i, j, k + 1);
199 m[2][2] = (m1 > m2) ? 1.0_rt : -1.0_rt;
200
201 youngs_normal(i, j, k, F, m[3][0], m[3][1], m[3][2]);
202 myc_choose<3>(m, mx, my, mz);
203}

228AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void myc_normal(const int i,

234 amrex::Real& mz) {
235 // m[c][d]: candidate c (0-1 centered height-function schemes along x, y,
236 // 2 Youngs), component d
237 amrex::Real m[3][3] = {{0.0_rt, 0.0_rt, 0.0_rt}, {0.0_rt, 0.0_rt, 0.0_rt}, {0.0_rt, 0.0_rt, 0.0_rt}};

239 // Column / row sums
240 const amrex::Real cL = F(i - 1, j - 1, k) + F(i - 1, j, k) + F(i - 1, j + 1, k);
241 const amrex::Real cR = F(i + 1, j - 1, k) + F(i + 1, j, k) + F(i + 1, j + 1, k);
242 const amrex::Real rB = F(i - 1, j - 1, k) + F(i, j - 1, k) + F(i + 1, j - 1, k);
243 const amrex::Real rT = F(i - 1, j + 1, k) + F(i, j + 1, k) + F(i + 1, j + 1, k);
244
245 // x as the height direction: sgn(mx) X = my Y + alpha
246 m[0][0] = (cL > cR) ? 1.0_rt : -1.0_rt;
247 m[0][1] = 0.5_rt * (rB - rT);
248 // y as the height direction: sgn(my) Y = mx X + alpha
249 m[1][0] = 0.5_rt * (cL - cR);
250 m[1][1] = (rB > rT) ? 1.0_rt : -1.0_rt;
251
252 youngs_normal(i, j, k, F, m[2][0], m[2][1], m[2][2]);
253 myc_choose<2>(m, mx, my, mz);
254}

257//! +1 if the liquid is on the low side of the column through iv along ed, else -1
258AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
259height_sigma(const amrex::IntVect& iv,
260 const amrex::IntVect& ed,
261 amrex::Array4<amrex::Real const> const& F) {
262 return (F(iv - hf_half * ed) > F(iv + hf_half * ed)) ? 1.0_rt : -1.0_rt;
263}

265//! The column through c along ed over the rows r - hf_half to r + hf_half: false unless F is
266//! monotone along it, from pure liquid to pure gas in the orientation sigma. H is the sum of F.
267AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool
268height_column(const amrex::IntVect& c,
269 const amrex::IntVect& ed,
270 const amrex::Real sigma,
271 const int r,
272 amrex::Array4<amrex::Real const> const& F,
273 amrex::Real& H) {
274 const auto liquid = [&](const int s) {
275 const amrex::Real f = F(c + s * ed);
276 return (sigma > 0.0_rt) ? f : 1.0_rt - f;
277 };
278 if (liquid(r - hf_half) < 1.0_rt - hf_pure_tol || liquid(r + hf_half) > hf_pure_tol) {
279 return false;
280 }
281 H = 0.0_rt;
282 amrex::Real prev = 1.0_rt;
283 for (int s = r - hf_half; s <= r + hf_half; ++s) {
284 const amrex::Real l = liquid(s);
285 if (l > prev + hf_pure_tol) {
286 return false;
287 }
288 prev = l;
289 H += F(c + s * ed);
290 }
291 return true;
292}

294/**
295 * \brief The row r, along d from (i,j,k), on which height_normal centers the neighboring columns:
296 * where the column through (i,j,k) crosses the interface. The crossing lies Lc - hf_half -
297 * 1/2 from the center of (i,j,k), with Lc the sum of the phase on the low side of the
298 * column, and within 1.5 as the cell is mixed: r is the nearest row, in [-1, 1].
299 * \return false if the column through (i,j,k) is not a height; sigma as height_sigma
300 */
301AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool
302height_row(const amrex::IntVect& iv,
303 const int d,
304 amrex::Array4<amrex::Real const> const& F,
305 int& r,
306 amrex::Real& sigma) {
307 const amrex::IntVect ed = amrex::IntVect::TheDimensionVector(d);
308 sigma = height_sigma(iv, ed, F);
309 amrex::Real Hc = 0.0_rt;
310 if (!height_column(iv, ed, sigma, 0, F, Hc)) {
311 return false;
312 }
313 const amrex::Real Lc = (sigma > 0.0_rt) ? Hc : static_cast<amrex::Real>(2 * hf_half + 1) - Hc;
314 r = amrex::max(-1, amrex::min(1, static_cast<int>(
315 std::floor(Lc - static_cast<amrex::Real>(hf_half)))));
316 return true;
317}

319/**
320 * \brief Height-function normal with height direction d: the liquid heights of the 7-cell columns
321 * through the tangential neighbors of (i,j,k), differenced centrally. The columns are
322 * centered on the row where the column through (i,j,k) crosses the interface, at most one
323 * row from it, so they read up to hf_half + 1 cells along d. Exact for a planar interface
324 * whose slopes relative to d are at most 1, which holds for the dominant normal direction:
325 * the crossing of a neighboring column then lies within 2.5 cells of that row, and its
326 * end cells are pure. Returns false unless each column, the center one
327 * included, crosses the interface once: from pure liquid to pure gas in the orientation of
328 * the center column, with F monotone in between. A thin gas gap with a liquid sliver inside
329 * a column is not a height, and the caller falls back to another normal.
330 */
331AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool
332height_normal(const int i,
333 const int j,
334 const int k,
335 const int d,
336 amrex::Array4<amrex::Real const> const& F,
337 amrex::Real m[3]) {
338 amrex::ignore_unused(k);
339 const amrex::IntVect iv(AMREX_D_DECL(i, j, k));
340 const amrex::IntVect ed = amrex::IntVect::TheDimensionVector(d);
341 // sigma: +1 for liquid on the low side
342 int r = 0;
343 amrex::Real sigma = 0.0_rt;
344 if (!height_row(iv, d, F, r, sigma)) {
345 return false;
346 }
347 m[0] = m[1] = m[2] = 0.0_rt;
348 for (int t = 0; t < AMREX_SPACEDIM; ++t) {
349 if (t == d) {
350 continue;
351 }
352 const amrex::IntVect et = amrex::IntVect::TheDimensionVector(t);
353 amrex::Real Hm = 0.0_rt;
354 amrex::Real Hp = 0.0_rt;
355 if (!height_column(iv - et, ed, sigma, r, F, Hm) ||
356 !height_column(iv + et, ed, sigma, r, F, Hp)) {
357 return false;
358 }
359 // m = grad(x_d - h) (sigma = 1) or grad(h - x_d): m_t = (H- - H+) / 2
360 m[t] = 0.5_rt * (Hm - Hp);
361 }
362 m[d] = sigma;
363 return true;
364}

369// To add one, write a struct like these and give it a name in dispatch_normal below. Nothing else
370// in the PLIC directory depends on which scheme is in use. A scheme has
371//
372// static constexpr int stencil;
373// The largest distance, in cells along any direction, from (i,j,k) that normal() reads. It
374// sets the ghost cells needed in F (PLIC::nGrowF).
375//
376// static void normal(i, j, k, F, mx, my, mz);
377// The normal in cell (i,j,k), pointing out of the liquid, in unit-cube cell coordinates, at any
378// scale, but not far below 1, as a sum|m| below machine epsilon counts as null. It is called
379// in mixed cells only, and not on structures one cell thick, where
380// fit_plane takes the principal direction of the Hessian of F instead (thin_structure,
381// degenerate_normal), as it does for a null normal.
382//
383// static int reach(i, j, k, dir, sgn, F, mm); (only with stencil > 1, with normal_from_myc)
384// The distance along direction dir, toward side sgn, that normal() reads in this cell, at most
385// stencil. It reads only cells within that distance, so that it holds whatever lies beyond.
386// The wall ghost cells take the plane of a cell near the wall that does not read ghost cells,
387// found with this (see wall_contact_planes). mm is the MYC normal of the cell (myc_normal),
388// which the wall fill computes once, for this and for normal_from_myc(i, j, k, F, mm, mx, my,
389// mz), which is normal() given mm.

447 //! normal(), given the MYC normal mm of the cell
448 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE static void
449 normal_from_myc(const int i,
450 const int j,
451 const int k,
452 amrex::Array4<amrex::Real const> const& F,
453 const amrex::Real mm[3],
454 amrex::Real& mx,
455 amrex::Real& my,
456 amrex::Real& mz) noexcept {
457 mx = mm[0];
458 my = mm[1];
459 mz = mm[2];
460 if (amrex::max(std::abs(mx), std::abs(my), std::abs(mz)) <= 0.0_rt) {
461 return;
462 }
463 amrex::Real m[3] = {0.0_rt, 0.0_rt, 0.0_rt};
464 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
465 if (!height_direction(mm, d)) {
466 continue;

468 amrex::Real mh[3];
469 if (!height_normal(i, j, k, d, F, mh)) {
470 return;
471 }
472 const amrex::Real norm = std::abs(mh[0]) + std::abs(mh[1]) + std::abs(mh[2]);
473 for (int a = 0; a < 3; ++a) {
474 m[a] += mh[a] / norm;
475 }
476 }
477 mx = m[0];
478 my = m[1];
479 mz = m[2];
480 }

516/**
517 * \brief Whether cell (i,j,k) lies on a structure one cell thick (a film, a gap, a filament or a
518 * drop), where the gradient is noise and the normal comes from the Hessian instead
519 * (degenerate_normal): along some line through it (an axis or a diagonal) its neighbors on
520 * both sides are pure in the same phase, and it differs from them by more than 8 times as
521 * much as F changes across it along any line. The margin keeps the gradient at the tip of
522 * an interface, as on spheres down to a radius of 0.6 cells.
523 */
524AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool thin_structure(const int i,
525 const int j,
526 const int k,
527 amrex::Array4<amrex::Real const> const& F) {
528 amrex::ignore_unused(k);
529 const amrex::IntVect iv(AMREX_D_DECL(i, j, k));
530 const amrex::Real fc = F(iv);
531 amrex::Real depth = 0.0_rt;
532 amrex::Real slope = 0.0_rt;
533 // The lines through the cell: the offsets e in {-1, 0, 1}^dim after the center in lexicographic
534 // order, one of each pair +-e
535 constexpr int noff = AMREX_D_TERM(3, *3, *3);
536 for (int o = noff / 2 + 1; o < noff; ++o) {
537 const amrex::IntVect e(AMREX_D_DECL(o % 3 - 1, (o / 3) % 3 - 1, o / 9 - 1));
538 const amrex::Real fp = F(iv + e);
539 const amrex::Real fm = F(iv - e);
540 // Pure as at the ends of a height column
541 const bool gas = fp <= hf_pure_tol && fm <= hf_pure_tol;
542 const bool liquid = fp >= 1.0_rt - hf_pure_tol && fm >= 1.0_rt - hf_pure_tol;
543 if (gas || liquid) {
544 depth = amrex::max(depth, amrex::min(std::abs(fp - fc), std::abs(fm - fc)));
545 }
546 slope = amrex::max(slope, 0.5_rt * std::abs(fp - fm));
547 }
548 return depth > 8.0_rt * slope;
549}

551/**
552 * \brief The orientation of a structure one cell thick (a film, a gap, a filament or a drop), where
553 * the normals vanish or are noise: across it, along the eigenvector of the
554 * largest eigenvalue (in magnitude) of the Hessian of F, from central differences. Where
555 * eigenvalues tie, the sum of their eigenvectors; for a structure aligned with the grid the
556 * Hessian is diagonal, and that is the sum of the tied axes. Each eigenvector is
557 * oriented by a fixed reference direction, so that the cells along a structure agree.
558 */
559AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void degenerate_normal(const int i,
560 const int j,
561 const int k,
562 amrex::Array4<amrex::Real const> const& F,
563 amrex::Real& mx,
564 amrex::Real& my,
565 amrex::Real& mz) {
566 amrex::ignore_unused(k);
567 const amrex::IntVect iv(AMREX_D_DECL(i, j, k));
568 constexpr int n = AMREX_SPACEDIM;
569 // The Hessian h, and the eigenvectors v (columns) of its Jacobi diagonalization
570 amrex::Real h[3][3] = {{0.0_rt, 0.0_rt, 0.0_rt}, {0.0_rt, 0.0_rt, 0.0_rt}, {0.0_rt, 0.0_rt, 0.0_rt}};
571 amrex::Real v[3][3] = {{1.0_rt, 0.0_rt, 0.0_rt}, {0.0_rt, 1.0_rt, 0.0_rt}, {0.0_rt, 0.0_rt, 1.0_rt}};
572 for (int a = 0; a < n; ++a) {
573 const amrex::IntVect ea = amrex::IntVect::TheDimensionVector(a);
574 h[a][a] = F(iv + ea) + F(iv - ea) - 2.0_rt * F(iv);
575 for (int b = a + 1; b < n; ++b) {
576 const amrex::IntVect eb = amrex::IntVect::TheDimensionVector(b);
577 h[a][b] = 0.25_rt * (F(iv + ea + eb) - F(iv + ea - eb) - F(iv - ea + eb) + F(iv - ea - eb));
578 h[b][a] = h[a][b];
579 }
580 }
581 constexpr amrex::Real eps = std::numeric_limits<amrex::Real>::epsilon();
582 bool rotated = true;
583 for (int sweep = 0; sweep < 10 && rotated; ++sweep) {
584 rotated = false;
585 for (int p = 0; p < n; ++p) {
586 for (int q = p + 1; q < n; ++q) {
587 if (std::abs(h[p][q]) <= eps * (std::abs(h[p][p]) + std::abs(h[q][q])) ||
588 std::abs(h[p][q]) <= std::numeric_limits<amrex::Real>::min()) {
589 continue;
590 }
591 rotated = true;
592 // The rotation in the (p, q) plane that zeroes h[p][q]
593 const amrex::Real theta = (h[q][q] - h[p][p]) / (2.0_rt * h[p][q]);
594 const amrex::Real t = ((theta >= 0.0_rt) ? 1.0_rt : -1.0_rt) /
595 (std::abs(theta) + std::sqrt(theta * theta + 1.0_rt));
596 const amrex::Real c = 1.0_rt / std::sqrt(t * t + 1.0_rt);
597 const amrex::Real sn = t * c;
598 // Over all three rows: in 2D the third is zero (of v, the identity row) and stays so
599 for (auto& row : h) {
600 const amrex::Real hp = row[p];
601 const amrex::Real hq = row[q];
602 row[p] = c * hp - sn * hq;
603 row[q] = sn * hp + c * hq;
604 }
605 for (int r = 0; r < n; ++r) {
606 const amrex::Real hp = h[p][r];
607 const amrex::Real hq = h[q][r];
608 h[p][r] = c * hp - sn * hq;
609 h[q][r] = sn * hp + c * hq;
610 }
611 for (auto& row : v) {
612 const amrex::Real vp = row[p];
613 const amrex::Real vq = row[q];
614 row[p] = c * vp - sn * vq;
615 row[q] = sn * vp + c * vq;
616 }
617 h[p][q] = 0.0_rt;
618 h[q][p] = 0.0_rt;
619 }
620 }
621 }
622 amrex::Real lmax = 0.0_rt;
623 for (int a = 0; a < n; ++a) {
624 lmax = amrex::max(lmax, std::abs(h[a][a]));
625 }
626 // Each tied eigenvector on the positive side of a reference direction, so that the sum does not
627 // cancel and the cells along a structure agree. Its components (1, sqrt(2) - 1, sqrt(3) - 3/2)
628 // are rationally independent, so no direction of integer components is normal to it.
629 constexpr amrex::Real ref[3] = {1.0_rt, 0.41421356237309515_rt, 0.2320508075688772_rt};
630 amrex::Real m[3] = {0.0_rt, 0.0_rt, 0.0_rt};
631 for (int a = 0; a < n; ++a) {
632 if (!tied(std::abs(h[a][a]), lmax)) {
633 continue;
634 }
635 amrex::Real proj = 0.0_rt;
636 for (int r = 0; r < n; ++r) {
637 proj += ref[r] * v[r][a];
638 }
639 const amrex::Real sgn = (proj >= 0.0_rt) ? 1.0_rt : -1.0_rt;
640 for (int r = 0; r < n; ++r) {
641 m[r] += sgn * v[r][a];
642 }
643 }
644 mx = m[0];
645 my = m[1];
646 mz = m[2];
647}

649/**
650 * \brief Normalize the normal (mx, my, mz) to sum|m| = 1 and give the constant alpha of the plane
651 * m.x = alpha, in unit-cube cell coordinates, that holds the volume fraction f of the cell.
652 */
653AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void plane_constant(const amrex::Real f,
654 amrex::Real& mx,
655 amrex::Real& my,
656 amrex::Real& mz,
657 amrex::Real& alpha) {
658 // A null normal (fit_plane replaces one by degenerate_normal): pick an arbitrary one
659 amrex::Real norm = std::abs(mx) + std::abs(my) + std::abs(mz);
660 if (norm <= std::numeric_limits<amrex::Real>::epsilon()) {
661 mx = 1.0_rt;
662 my = 0.0_rt;
663 mz = 0.0_rt;
664 norm = 1.0_rt;
665 }
666 mx /= norm;
667 my /= norm;
668 mz /= norm;
669
670 alpha = volume_intercept(std::abs(mx), std::abs(my), std::abs(mz), f);
671
672 // Back from the reflected (all positive) frame to the cell frame
673 alpha += amrex::min(0.0_rt, mx) + amrex::min(0.0_rt, my) + amrex::min(0.0_rt, mz);
674}

723/**
724 * \brief Normal of an interface that runs inside the wall layer, from a one-sided height function:
725 * with the cell inside the wall layer pure, and the ghost cell beyond it taken as pure of the
726 * other phase, the volume fraction of that phase in a wall cell is the height of the
727 * interface at the cell center, measured from the wall face, and centered differences of
728 * those heights along the wall give its slopes. Exact for a planar interface that stays
729 * within the wall layer across the cell and its tangential neighbors, which is checked: the
730 * layer inside must be pure, of one phase, and each height plus or minus half the change of
731 * height across a cell must lie in [0, 1]. Reads the wall layer and the layer inside it,
732 * one cell away along the wall, and no ghost cells.
733 *
734 * \param c the wall cell
735 * \param sgn -1 for a low wall, +1 for a high wall
736 * \return false where the height function does not hold
737 */
738AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool
739wall_layer_normal(const amrex::IntVect& c,
740 const int dir,
741 const int sgn,
742 amrex::Array4<amrex::Real const> const& F,
743 amrex::Real* m) {
744 const amrex::IntVect e_in = (-sgn) * amrex::IntVect::TheDimensionVector(dir);
745 const amrex::Real f_in = F(c + e_in);
746 if (is_mixed(f_in)) {
747 return false;
748 }
749 const bool liquid_in = f_in >= 0.5_rt;
750 // The tangential directions in increasing order, with a dummy one in 2D
751 const int t[2] = {(dir == 0) ? 1 : 0, (AMREX_SPACEDIM == 3) ? ((dir == 2) ? 1 : 2) : 0};
752 const int b_max = (AMREX_SPACEDIM == 3) ? 1 : 0;
753 // Heights of the wall-side phase, h[b+1][a+1] at offsets a along t[0] and b along t[1]
754 amrex::Real h[3][3] = {{0.0_rt, 0.0_rt, 0.0_rt}, {0.0_rt, 0.0_rt, 0.0_rt}, {0.0_rt, 0.0_rt, 0.0_rt}};
755 for (int b = -b_max; b <= b_max; ++b) {
756 for (int a = -1; a <= 1; ++a) {
757 amrex::IntVect cc = c;
758 cc[t[0]] += a;
759 if (AMREX_SPACEDIM == 3) {
760 cc[t[1]] += b;
761 }
762 const amrex::Real f = F(cc);
763 const amrex::Real p = F(cc + e_in);
764 if (!is_mixed(f) || is_mixed(p) || ((p >= 0.5_rt) != liquid_in)) {
765 return false;
766 }
767 h[b + 1][a + 1] = liquid_in ? 1.0_rt - f : f;
768 }
769 }
770 const amrex::Real s0 = 0.5_rt * (h[1][2] - h[1][0]);
771 const amrex::Real s1 = (AMREX_SPACEDIM == 3) ? 0.5_rt * (h[2][1] - h[0][1]) : 0.0_rt;
772 const amrex::Real spread = 0.5_rt * (std::abs(s0) + std::abs(s1));
773 for (int b = -b_max; b <= b_max; ++b) {
774 for (int a = -1; a <= 1; ++a) {
775 const amrex::Real hc = h[b + 1][a + 1];
776 if (hc - spread < 0.0_rt || hc + spread > 1.0_rt) {
777 return false;
778 }
779 }
780 }
781 // m.x - alpha < 0 in the liquid: the liquid is inside the interface if liquid_in
782 const amrex::Real sigma = liquid_in ? 1.0_rt : -1.0_rt;
783 m[0] = m[1] = m[2] = 0.0_rt;
784 m[t[0]] = sigma * s0;
785 if (AMREX_SPACEDIM == 3) {
786 m[t[1]] = sigma * s1;
787 }
788 m[dir] = sigma * static_cast<amrex::Real>(sgn);
789 return true;
790}

817//! Call f(offset, distance) for the offsets along the wall (zero along dir) within wall_nsearch cells
818//! in each tangential direction, with distance the sum of their absolute components, by increasing
819//! distance. Stops when f returns false.
820template <typename Fn>
821AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void for_each_wall_offset(const int dir, Fn const& f) {
822 constexpr int ns = wall_nsearch;
823#if AMREX_SPACEDIM == 3
824 // The tangential directions
825 const int to = (dir == 2) ? 1 : 2;
826 const int ti = (dir == 0) ? 1 : 0;
827 for (int t = 0; t <= 2 * ns; ++t) {
828 for (int b = -ns; b <= ns; ++b) {

830 if (rem < 0 || rem > ns) {
831 continue;
832 }
833 // -rem, then +rem unless it is the same

837 off[to] = b;
838 if (!f(off, t)) {
839 return;
840 }
841 }
842 }
843 }
844#else

846 for (int t = 0; t <= ns; ++t) {
847 for (int a = -t; a <= t; a += amrex::max(1, 2 * t)) {
848 amrex::IntVect off(0);
849 off[ti] = a;

851 return;
852 }
853 }
854 }
855#endif

858//! What a cell offers the search along a wall (see wall_candidate)
859constexpr int wc_none = 0; //!< nothing
860constexpr int wc_myc = 1; //!< below the wall layer, mixed: its MYC plane reads only data and no ghost cells
861constexpr int wc_plane = 2; //!< as wc_myc, and the plane of the normal scheme does too
862constexpr int wc_height = 3; //!< in the wall layer, where the height function of wall_layer_normal holds
863
864/**
865 * \brief What cell c, up to Normal::stencil layers inside the wall, offers the search of
866 * wall_contact_planes, and the plane it offers in m and alpha. Computed once per cell, as the
867 * searches of neighboring wall cells overlap.
868 *
869 * \param fit_near nonzero on the cells of F whose 3x3(x3) block holds only data
870 * \param fit_ok nonzero on the cells of F whose reconstruction stencil reads only data
871 */
872template <typename Normal>
873AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE int
874wall_candidate(const amrex::IntVect& c,
875 const int dir,
876 const int iwall,
877 const int sgn,
878 amrex::Array4<int const> const& fit_near,
879 amrex::Array4<int const> const& fit_ok,
880 amrex::Array4<amrex::Real const> const& F,
881 amrex::Real* m,
882 amrex::Real& alpha) {
883 if (!fit_near.contains(c) || fit_near(c) == 0) {
884 return wc_none;
885 }
886 const int layer = (iwall - c[dir]) * sgn;
887 if (layer == 0) {
888 if (!wall_layer_normal(c, dir, sgn, F, m)) {
889 return wc_none;
890 }
891 plane_constant(F(c), m[0], m[1], m[2], alpha);
892 return wc_height;
893 }
894 if (!is_mixed(F(c))) {
895 return wc_none;
896 }
897 // The stencil of the scheme would read cells without data, or wall-normal ghost cells
898 const auto cell = c.dim3();
899 if constexpr (Normal::stencil > 1) {
900 // The MYC normal once, for the reach, and for either plane
901 amrex::Real mm[3];
902 myc_normal(cell.x, cell.y, cell.z, F, mm[0], mm[1], mm[2]);
903 if (fit_ok(c) == 0 ||
904 (layer < Normal::stencil && Normal::reach(cell.x, cell.y, cell.z, dir, sgn, F, mm) > layer)) {
905 fit_plane_with(cell.x, cell.y, cell.z, F,
906 [&] (amrex::Real& x, amrex::Real& y, amrex::Real& z) { x = mm[0]; y = mm[1]; z = mm[2]; },
907 m[0], m[1], m[2], alpha);
908 return wc_myc;
909 }
910 fit_plane_with(cell.x, cell.y, cell.z, F,
911 [&] (amrex::Real& x, amrex::Real& y, amrex::Real& z) {
912 Normal::normal_from_myc(cell.x, cell.y, cell.z, F, mm, x, y, z);
913 },
914 m[0], m[1], m[2], alpha);
915 return wc_plane;
916 }
917 fit_plane<Normal>(cell.x, cell.y, cell.z, F, m[0], m[1], m[2], alpha);
918 return wc_plane;
919}

921/**
922 * \brief The PLIC planes that the ghost cells beyond the wall cell iv_w extend, when the interface
923 * meets the wall within wall_nsearch cells (tangentially) of iv_w: a free contact line,
924 * exact for a planar interface and for one uniform normal to the wall. The planes are those
925 * of the nearest interface cells, up to Normal::stencil layers inside the wall, whose
926 * reconstruction does not read ghost cells (Normal::reach). Below about 26.6 degrees from
927 * the wall, or near another wall, where myc_height finds none, the MYC planes of the nearest
928 * interface cells below the wall layer, which are exact there (see wall_nsearch), down to about
929 * atan(1/wall_nsearch). Below that, the interface runs inside the wall layer near the
930 * contact line, and the planes are those of the nearest wall cells where the one-sided
931 * height function of wall_layer_normal holds. All the equally near cells give their plane,
932 * so that no direction is preferred. No plane reads ghost cells, so one pass of the fill is
933 * enough.
934 *
935 * \tparam Normal the normal scheme
936 * \param iwall index along dir of the wall-adjacent interior cell
937 * \param sgn -1 for a low wall, +1 for a high wall
938 * \param cand wall_candidate of the cells searched
939 * \param pl the plane each candidate offers (m_x, m_y, m_z, alpha)
940 * \param m the sum of the planes taken, in the coordinates of iv_w, with alpha
941 * \return the number of planes
942 */
943template <typename Normal>
944AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE int
945wall_contact_planes(const amrex::IntVect& iv_w,
946 const int dir,
947 const int iwall,
948 const int sgn,
949 amrex::Array4<int const> const& cand,
950 amrex::Array4<amrex::Real const> const& pl,
951 amrex::Real* m,
952 amrex::Real& alpha) {
953 constexpr int nfar = 1000;
954
955 // The sum of the planes of the cells at the nearest distance found so far
956 struct Sum {
957 amrex::Real m[3] = {0.0_rt, 0.0_rt, 0.0_rt};
958 amrex::Real alpha = 0.0_rt;
959 int n = 0;
960 int dist = nfar;
961 // Only the nearest cells count
962 AMREX_GPU_HOST_DEVICE void add (amrex::IntVect const& iv, amrex::IntVect const& ref,
963 amrex::Array4<amrex::Real const> const& p, int d_ref) {
964 if (d_ref > dist) {
965 return;
966 }
967 if (d_ref < dist) {
968 *this = Sum{};
969 dist = d_ref;
970 }
971 alpha += p(ref, 3);
972 for (int d = 0; d < 3; ++d) {
973 m[d] += p(ref, d);
974 }
975 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
976 alpha -= p(ref, d) * static_cast<amrex::Real>(iv[d] - ref[d]);

1009 }
1010 }

1012 plane.add(iv_w, c, pl, dist);
1013 }
1014 }
1015 return true;
1016 });

1018 return take(plane);
1019 }
1020 if (any.n > 0) {
1021 return take(any);
1022 }
