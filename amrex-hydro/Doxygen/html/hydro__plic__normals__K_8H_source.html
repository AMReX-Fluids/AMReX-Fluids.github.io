30// Half-length of the height-function columns (7 cells)
31static constexpr int hf_half = 3;
32// A height column ends in cells within this of 0 or 1
33#ifdef AMREX_USE_FLOAT
34static constexpr amrex::Real hf_pure_tol = 1.0e-3_rt;
35#else
36static constexpr amrex::Real hf_pure_tol = 1.0e-6_rt;
37#endif
38// Relative difference below which two values tie, to round-off (see tied)
39static constexpr amrex::Real tie_tol = 1000.0_rt * std::numeric_limits<amrex::Real>::epsilon();
40
41//! Whether a ties with the largest of its kind amax (to round-off)
42AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool tied(const amrex::Real a, const amrex::Real amax) noexcept {
43 return a >= (1.0_rt - tie_tol) * amax;
44}

46/**
47 * \brief The MYC normal from its centered candidates m[c] (height direction c, c < n, height
48 * component +-1) and the Youngs normal m[n] (at any scale): the centered candidate whose
49 * height component dominates after L1 normalization, or the mean of those that tie, so that
50 * no direction is preferred, unless Youngs' largest component is smaller (an interface not
51 * aligned with a grid plane).
52 */
53template <int n>
54AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void myc_choose(amrex::Real (&m)[n + 1][3],
55 amrex::Real& mx,
56 amrex::Real& my,
57 amrex::Real& mz) {
58 // The L1 norm of each candidate, and the components compared after normalization: only the
59 // candidates kept are normalized in full
60 amrex::Real t[n + 1];
61 amrex::Real hc[n];
62 amrex::Real t0 = 0.0_rt;
63 for (int c = 0; c <= n; ++c) {
64 t[c] = std::abs(m[c][0]) + std::abs(m[c][1]) + std::abs(m[c][2]);
65 if (c < n) {
66 hc[c] = std::abs(m[c][c]) / t[c];
67 t0 = amrex::max(t0, hc[c]);
68 }
69 }
70 const amrex::Real ty =
71 (t[n] > 0.0_rt)
72 ? amrex::max(std::abs(m[n][0]), std::abs(m[n][1]), std::abs(m[n][2])) / t[n]
73 : 0.0_rt;
74 amrex::Real mc[3] = {0.0_rt, 0.0_rt, 0.0_rt};
75 if (ty > 0.0_rt && t0 > ty) {
76 for (int d = 0; d < 3; ++d) {
77 mc[d] = m[n][d] / t[n];
78 }
79 } else {
80 for (int c = 0; c < n; ++c) {
81 if (tied(hc[c], t0)) {
82 for (int d = 0; d < 3; ++d) {
83 mc[d] += m[c][d] / t[c];
84 }
85 }
86 }
87 }
88 // Tied candidates may cancel: a null normal, which fit_plane replaces
89 const amrex::Real tc = std::abs(mc[0]) + std::abs(mc[1]) + std::abs(mc[2]);
90 const amrex::Real scale = (tc > 0.0_rt) ? 1.0_rt / tc : 0.0_rt;
91 mx = mc[0] * scale;
92 my = mc[1] * scale;
93 mz = mc[2] * scale;
94}

97/**
98 * \brief Youngs' finite-difference normal on the 27-point stencil, scaled
99 * by -32 (i.e. m = -32 grad F).
100 */
101AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void
102youngs_normal(const int i,
103 const int j,
104 const int k,
105 amrex::Array4<amrex::Real const> const& F,
106 amrex::Real& mx,
107 amrex::Real& my,
108 amrex::Real& mz) {
109 amrex::Real mm1 = F(i - 1, j - 1, k - 1) + F(i - 1, j - 1, k + 1) + F(i - 1, j + 1, k - 1) +
110 F(i - 1, j + 1, k + 1) +
111 2.0_rt * (F(i - 1, j - 1, k) + F(i - 1, j + 1, k) + F(i - 1, j, k - 1) +
112 F(i - 1, j, k + 1)) +
113 4.0_rt * F(i - 1, j, k);
114 amrex::Real mm2 = F(i + 1, j - 1, k - 1) + F(i + 1, j - 1, k + 1) + F(i + 1, j + 1, k - 1) +
115 F(i + 1, j + 1, k + 1) +
116 2.0_rt * (F(i + 1, j - 1, k) + F(i + 1, j + 1, k) + F(i + 1, j, k - 1) +
117 F(i + 1, j, k + 1)) +
118 4.0_rt * F(i + 1, j, k);
119 mx = mm1 - mm2;
120
121 mm1 = F(i - 1, j - 1, k - 1) + F(i - 1, j - 1, k + 1) + F(i + 1, j - 1, k - 1) +
122 F(i + 1, j - 1, k + 1) +
123 2.0_rt *
124 (F(i - 1, j - 1, k) + F(i + 1, j - 1, k) + F(i, j - 1, k - 1) + F(i, j - 1, k + 1)) +
125 4.0_rt * F(i, j - 1, k);
126 mm2 = F(i - 1, j + 1, k - 1) + F(i - 1, j + 1, k + 1) + F(i + 1, j + 1, k - 1) +
127 F(i + 1, j + 1, k + 1) +
128 2.0_rt *
129 (F(i - 1, j + 1, k) + F(i + 1, j + 1, k) + F(i, j + 1, k - 1) + F(i, j + 1, k + 1)) +
130 4.0_rt * F(i, j + 1, k);
131 my = mm1 - mm2;
132
133 mm1 = F(i - 1, j - 1, k - 1) + F(i - 1, j + 1, k - 1) + F(i + 1, j - 1, k - 1) +
134 F(i + 1, j + 1, k - 1) +
135 2.0_rt *
136 (F(i - 1, j, k - 1) + F(i + 1, j, k - 1) + F(i, j - 1, k - 1) + F(i, j + 1, k - 1)) +
137 4.0_rt * F(i, j, k - 1);
138 mm2 = F(i - 1, j - 1, k + 1) + F(i - 1, j + 1, k + 1) + F(i + 1, j - 1, k + 1) +
139 F(i + 1, j + 1, k + 1) +
140 2.0_rt *
141 (F(i - 1, j, k + 1) + F(i + 1, j, k + 1) + F(i, j - 1, k + 1) + F(i, j + 1, k + 1)) +
142 4.0_rt * F(i, j, k + 1);
143 mz = mm1 - mm2;
144}

146/**
147 * \brief Mixed Youngs-Centered normal (3D). Returns an L1-normalized
148 * normal pointing out of the liquid.
149 */
150AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void myc_normal(const int i,
151 const int j,
152 const int k,
153 amrex::Array4<amrex::Real const> const& F,
154 amrex::Real& mx,
155 amrex::Real& my,
156 amrex::Real& mz) {
157 // m[c][d]: candidate c (0-2 centered height-function schemes along x, y,
158 // z, 3 Youngs), component d
159 amrex::Real m[4][3];
160
161 // Centered scheme with x as the height direction:
162 // sgn(mx) X = my Y + mz Z + alpha
163 amrex::Real m1 = F(i - 1, j, k - 1) + F(i - 1, j, k + 1) + F(i - 1, j - 1, k) +
164 F(i - 1, j + 1, k) + F(i - 1, j, k);
165 amrex::Real m2 = F(i + 1, j, k - 1) + F(i + 1, j, k + 1) + F(i + 1, j - 1, k) +
166 F(i + 1, j + 1, k) + F(i + 1, j, k);
167 m[0][0] = (m1 > m2) ? 1.0_rt : -1.0_rt;
168 m1 = F(i - 1, j - 1, k) + F(i + 1, j - 1, k) + F(i, j - 1, k);
169 m2 = F(i - 1, j + 1, k) + F(i + 1, j + 1, k) + F(i, j + 1, k);
170 m[0][1] = 0.5_rt * (m1 - m2);
171 m1 = F(i - 1, j, k - 1) + F(i + 1, j, k - 1) + F(i, j, k - 1);
172 m2 = F(i - 1, j, k + 1) + F(i + 1, j, k + 1) + F(i, j, k + 1);
173 m[0][2] = 0.5_rt * (m1 - m2);
174
175 // y as the height direction
176 m1 = F(i - 1, j - 1, k) + F(i - 1, j + 1, k) + F(i - 1, j, k);
177 m2 = F(i + 1, j - 1, k) + F(i + 1, j + 1, k) + F(i + 1, j, k);
178 m[1][0] = 0.5_rt * (m1 - m2);
179 m1 = F(i, j - 1, k - 1) + F(i, j - 1, k + 1) + F(i + 1, j - 1, k) + F(i - 1, j - 1, k) +
180 F(i, j - 1, k);
181 m2 = F(i, j + 1, k - 1) + F(i, j + 1, k + 1) + F(i + 1, j + 1, k) + F(i - 1, j + 1, k) +
182 F(i, j + 1, k);
183 m[1][1] = (m1 > m2) ? 1.0_rt : -1.0_rt;
184 m1 = F(i, j - 1, k - 1) + F(i, j, k - 1) + F(i, j + 1, k - 1);
185 m2 = F(i, j - 1, k + 1) + F(i, j, k + 1) + F(i, j + 1, k + 1);
186 m[1][2] = 0.5_rt * (m1 - m2);
187
188 // z as the height direction
189 m1 = F(i - 1, j, k - 1) + F(i - 1, j, k + 1) + F(i - 1, j, k);
190 m2 = F(i + 1, j, k - 1) + F(i + 1, j, k + 1) + F(i + 1, j, k);
191 m[2][0] = 0.5_rt * (m1 - m2);
192 m1 = F(i, j - 1, k - 1) + F(i, j - 1, k + 1) + F(i, j - 1, k);
193 m2 = F(i, j + 1, k - 1) + F(i, j + 1, k + 1) + F(i, j + 1, k);
194 m[2][1] = 0.5_rt * (m1 - m2);
195 m1 = F(i - 1, j, k - 1) + F(i + 1, j, k - 1) + F(i, j - 1, k - 1) + F(i, j + 1, k - 1) +
196 F(i, j, k - 1);
197 m2 = F(i - 1, j, k + 1) + F(i + 1, j, k + 1) + F(i, j - 1, k + 1) + F(i, j + 1, k + 1) +
198 F(i, j, k + 1);
199 m[2][2] = (m1 > m2) ? 1.0_rt : -1.0_rt;
200
201 youngs_normal(i, j, k, F, m[3][0], m[3][1], m[3][2]);
202 myc_choose<3>(m, mx, my, mz);
203}

224/**
225 * \brief Mixed Youngs-Centered normal (2D). Returns an L1-normalized
226 * normal pointing out of the liquid.
227 */
228AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void myc_normal(const int i,
229 const int j,
230 const int k,
231 amrex::Array4<amrex::Real const> const& F,
232 amrex::Real& mx,
233 amrex::Real& my,
234 amrex::Real& mz) {
235 // m[c][d]: candidate c (0-1 centered height-function schemes along x, y,
236 // 2 Youngs), component d
237 amrex::Real m[3][3] = {{0.0_rt, 0.0_rt, 0.0_rt}, {0.0_rt, 0.0_rt, 0.0_rt}, {0.0_rt, 0.0_rt, 0.0_rt}};
238
239 // Column / row sums
240 const amrex::Real cL = F(i - 1, j - 1, k) + F(i - 1, j, k) + F(i - 1, j + 1, k);
241 const amrex::Real cR = F(i + 1, j - 1, k) + F(i + 1, j, k) + F(i + 1, j + 1, k);
242 const amrex::Real rB = F(i - 1, j - 1, k) + F(i, j - 1, k) + F(i + 1, j - 1, k);
243 const amrex::Real rT = F(i - 1, j + 1, k) + F(i, j + 1, k) + F(i + 1, j + 1, k);
244
245 // x as the height direction: sgn(mx) X = my Y + alpha
246 m[0][0] = (cL > cR) ? 1.0_rt : -1.0_rt;
247 m[0][1] = 0.5_rt * (rB - rT);
248 // y as the height direction: sgn(my) Y = mx X + alpha
249 m[1][0] = 0.5_rt * (cL - cR);
250 m[1][1] = (rB > rT) ? 1.0_rt : -1.0_rt;
251
252 youngs_normal(i, j, k, F, m[2][0], m[2][1], m[2][2]);
253 myc_choose<2>(m, mx, my, mz);
254}

257/**
258 * \brief Height-function normal with height direction d: the liquid heights of the 7-cell columns
259 * through the tangential neighbors of (i,j,k), differenced centrally. The columns are
260 * centered on the row where the column through (i,j,k) crosses the interface, at most one
261 * row from it, so they read up to hf_half + 1 cells along d. Exact for a planar interface
262 * whose slopes relative to d are at most 1, which holds for the dominant normal direction:
263 * the crossing of a neighboring column then lies within 2.5 cells of that row, and its
264 * end cells are pure. Returns false unless each column, the center one
265 * included, crosses the interface once: from pure liquid to pure gas in the orientation of
266 * the center column, with F monotone in between. A thin gas gap with a liquid sliver inside
267 * a column is not a height, and the caller falls back to another normal.
268 */
269AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool
270height_normal(const int i,
271 const int j,
272 const int k,
273 const int d,
274 amrex::Array4<amrex::Real const> const& F,
275 amrex::Real m[3]) {
276 amrex::ignore_unused(k);
277 const amrex::IntVect iv(AMREX_D_DECL(i, j, k));
278 const amrex::IntVect ed = amrex::IntVect::TheDimensionVector(d);
279 // +1 for liquid on the low side
280 const amrex::Real sigma = (F(iv - hf_half * ed) > F(iv + hf_half * ed)) ? 1.0_rt : -1.0_rt;
281 // The column through c over the rows r - hf_half to r + hf_half: false unless F is monotone
282 // along it, from pure liquid to pure gas. H is the sum of F.
283 const auto column = [&](const amrex::IntVect& c, const int r, amrex::Real& H) {
284 const auto liquid = [&](const int s) {
285 const amrex::Real f = F(c + s * ed);
286 return (sigma > 0.0_rt) ? f : 1.0_rt - f;
287 };
288 if (liquid(r - hf_half) < 1.0_rt - hf_pure_tol || liquid(r + hf_half) > hf_pure_tol) {
289 return false;
290 }
291 H = 0.0_rt;
292 amrex::Real prev = 1.0_rt;
293 for (int s = r - hf_half; s <= r + hf_half; ++s) {
294 const amrex::Real l = liquid(s);
295 if (l > prev + hf_pure_tol) {
296 return false;
297 }
298 prev = l;
299 H += F(c + s * ed);
300 }
301 return true;
302 };
303 amrex::Real Hc = 0.0_rt;
304 if (!column(iv, 0, Hc)) {
305 return false;
306 }
307 // The column crosses Lc - hf_half - 1/2 along d from the center of (i,j,k), with Lc the sum of
308 // the phase on its low side, and within 1.5 as the cell is mixed: the nearest row
309 const amrex::Real Lc = (sigma > 0.0_rt) ? Hc : static_cast<amrex::Real>(2 * hf_half + 1) - Hc;
310 const int r = amrex::max(-1, amrex::min(1, static_cast<int>(
311 std::floor(Lc - static_cast<amrex::Real>(hf_half)))));
312 m[0] = m[1] = m[2] = 0.0_rt;
313 for (int t = 0; t < AMREX_SPACEDIM; ++t) {
314 if (t == d) {
315 continue;
316 }
317 const amrex::IntVect et = amrex::IntVect::TheDimensionVector(t);
318 amrex::Real Hm = 0.0_rt;
319 amrex::Real Hp = 0.0_rt;
320 if (!column(iv - et, r, Hm) || !column(iv + et, r, Hp)) {
321 return false;
322 }
323 // m = grad(x_d - h) (sigma = 1) or grad(h - x_d): m_t = (H- - H+) / 2
324 m[t] = 0.5_rt * (Hm - Hp);
325 }
326 m[d] = sigma;
327 return true;
328}

332//
333// To add one, write a struct like these and give it a name in dispatch_normal below. Nothing else
334// in the PLIC directory depends on which scheme is in use. A scheme has
335//
336// static constexpr int stencil;
337// The largest distance, in cells along any direction, from (i,j,k) that normal() reads. It
338// sets the ghost cells needed in F (PLIC::nGrowF).

341// The normal in cell (i,j,k), pointing out of the liquid, in unit-cube cell coordinates, at any
342// scale, but not far below 1, as a sum|m| below machine epsilon counts as null. It is called

344// fit_plane takes the principal direction of the Hessian of F instead (thin_structure,
345// degenerate_normal), as it does for a null normal.

380/**
381 * \brief The height-function normal along the dominant direction of the Mixed Youngs-Centered
382 * normal, or the latter where the heights are not defined (thin or strongly curved
383 * structures). Exact for a planar interface. Where components tie, the mean over the tied
384 * directions, so that no direction is preferred (the slopes are then 1 relative to each,
385 * where the heights are exact), and the MYC normal unless the heights hold along each.
386 */
387struct HeightFunctionMYC {
388 static constexpr int stencil = hf_half + 1;
389
390 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE static void
391 normal(const int i,
392 const int j,
393 const int k,
394 amrex::Array4<amrex::Real const> const& F,
395 amrex::Real& mx,
396 amrex::Real& my,
397 amrex::Real& mz) noexcept {
398 myc_normal(i, j, k, F, mx, my, mz);
399 const amrex::Real am[3] = {std::abs(mx), std::abs(my), std::abs(mz)};
400 const amrex::Real amax = amrex::max(am[0], am[1], am[2]);
401 if (amax <= 0.0_rt) {
402 return;
403 }
404 amrex::Real m[3] = {0.0_rt, 0.0_rt, 0.0_rt};
405 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
406 if (!tied(am[d], amax)) {
407 continue;
408 }
409 amrex::Real mh[3];
410 if (!height_normal(i, j, k, d, F, mh)) {
411 return;
412 }
413 const amrex::Real norm = std::abs(mh[0]) + std::abs(mh[1]) + std::abs(mh[2]);
414 for (int a = 0; a < 3; ++a) {
415 m[a] += mh[a] / norm;
416 }
417 }
418 mx = m[0];
419 my = m[1];
420 mz = m[2];
421 }
422};

424/**
425 * \brief Whether cell (i,j,k) lies on a structure one cell thick (a film, a gap, a filament or a
426 * drop), where the gradient is noise and the normal comes from the Hessian instead
427 * (degenerate_normal): along some line through it (an axis or a diagonal) its neighbors on
428 * both sides are pure in the same phase, and it differs from them by more than 8 times as
429 * much as F changes across it along any line. The margin keeps the gradient at the tip of
430 * an interface, as on spheres down to a radius of 0.6 cells.
431 */
432AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool thin_structure(const int i,
433 const int j,
434 const int k,
435 amrex::Array4<amrex::Real const> const& F) {
436 amrex::ignore_unused(k);
437 const amrex::IntVect iv(AMREX_D_DECL(i, j, k));
438 const amrex::Real fc = F(iv);
439 amrex::Real depth = 0.0_rt;
440 amrex::Real slope = 0.0_rt;
441 // The lines through the cell: the offsets e in {-1, 0, 1}^dim after the center in lexicographic
442 // order, one of each pair +-e
443 constexpr int noff = AMREX_D_TERM(3, *3, *3);
444 for (int o = noff / 2 + 1; o < noff; ++o) {
445 const amrex::IntVect e(AMREX_D_DECL(o % 3 - 1, (o / 3) % 3 - 1, o / 9 - 1));
446 const amrex::Real fp = F(iv + e);
447 const amrex::Real fm = F(iv - e);
448 // Pure as at the ends of a height column
449 const bool gas = fp <= hf_pure_tol && fm <= hf_pure_tol;
450 const bool liquid = fp >= 1.0_rt - hf_pure_tol && fm >= 1.0_rt - hf_pure_tol;
451 if (gas || liquid) {
452 depth = amrex::max(depth, amrex::min(std::abs(fp - fc), std::abs(fm - fc)));
453 }
454 slope = amrex::max(slope, 0.5_rt * std::abs(fp - fm));
455 }
456 return depth > 8.0_rt * slope;
457}

459/**
460 * \brief The orientation of a structure one cell thick (a film, a gap, a filament or a drop), where
461 * the normals vanish or are noise: across it, along the eigenvector of the
462 * largest eigenvalue (in magnitude) of the Hessian of F, from central differences. Where
463 * eigenvalues tie, the sum of their eigenvectors; for a structure aligned with the grid the
464 * Hessian is diagonal, and that is the sum of the tied axes. Each eigenvector is
465 * oriented by a fixed reference direction, so that the cells along a structure agree.
466 */
467AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void degenerate_normal(const int i,
468 const int j,
469 const int k,
470 amrex::Array4<amrex::Real const> const& F,
471 amrex::Real& mx,
472 amrex::Real& my,
473 amrex::Real& mz) {
474 amrex::ignore_unused(k);
475 const amrex::IntVect iv(AMREX_D_DECL(i, j, k));
476 constexpr int n = AMREX_SPACEDIM;
477 // The Hessian h, and the eigenvectors v (columns) of its Jacobi diagonalization
478 amrex::Real h[3][3] = {{0.0_rt, 0.0_rt, 0.0_rt}, {0.0_rt, 0.0_rt, 0.0_rt}, {0.0_rt, 0.0_rt, 0.0_rt}};
479 amrex::Real v[3][3] = {{1.0_rt, 0.0_rt, 0.0_rt}, {0.0_rt, 1.0_rt, 0.0_rt}, {0.0_rt, 0.0_rt, 1.0_rt}};
480 for (int a = 0; a < n; ++a) {
481 const amrex::IntVect ea = amrex::IntVect::TheDimensionVector(a);
482 h[a][a] = F(iv + ea) + F(iv - ea) - 2.0_rt * F(iv);
483 for (int b = a + 1; b < n; ++b) {
484 const amrex::IntVect eb = amrex::IntVect::TheDimensionVector(b);
485 h[a][b] = 0.25_rt * (F(iv + ea + eb) - F(iv + ea - eb) - F(iv - ea + eb) + F(iv - ea - eb));
486 h[b][a] = h[a][b];
487 }
488 }
489 constexpr amrex::Real eps = std::numeric_limits<amrex::Real>::epsilon();
490 bool rotated = true;
491 for (int sweep = 0; sweep < 10 && rotated; ++sweep) {
492 rotated = false;
493 for (int p = 0; p < n; ++p) {
494 for (int q = p + 1; q < n; ++q) {
495 if (std::abs(h[p][q]) <= eps * (std::abs(h[p][p]) + std::abs(h[q][q])) ||
496 std::abs(h[p][q]) <= std::numeric_limits<amrex::Real>::min()) {
497 continue;
498 }
499 rotated = true;
500 // The rotation in the (p, q) plane that zeroes h[p][q]
501 const amrex::Real theta = (h[q][q] - h[p][p]) / (2.0_rt * h[p][q]);
502 const amrex::Real t = ((theta >= 0.0_rt) ? 1.0_rt : -1.0_rt) /
503 (std::abs(theta) + std::sqrt(theta * theta + 1.0_rt));
504 const amrex::Real c = 1.0_rt / std::sqrt(t * t + 1.0_rt);
505 const amrex::Real sn = t * c;
506 // Over all three rows: in 2D the third is zero (of v, the identity row) and stays so
507 for (auto& row : h) {
508 const amrex::Real hp = row[p];
509 const amrex::Real hq = row[q];
510 row[p] = c * hp - sn * hq;
511 row[q] = sn * hp + c * hq;
512 }
513 for (int r = 0; r < n; ++r) {
514 const amrex::Real hp = h[p][r];
515 const amrex::Real hq = h[q][r];
516 h[p][r] = c * hp - sn * hq;
517 h[q][r] = sn * hp + c * hq;
518 }
519 for (auto& row : v) {
520 const amrex::Real vp = row[p];
521 const amrex::Real vq = row[q];
522 row[p] = c * vp - sn * vq;
523 row[q] = sn * vp + c * vq;
524 }
525 h[p][q] = 0.0_rt;
526 h[q][p] = 0.0_rt;
527 }
528 }
529 }
530 amrex::Real lmax = 0.0_rt;
531 for (int a = 0; a < n; ++a) {
532 lmax = amrex::max(lmax, std::abs(h[a][a]));
533 }
534 // Each tied eigenvector on the positive side of a reference direction, so that the sum does not
535 // cancel and the cells along a structure agree. Its components (1, sqrt(2) - 1, sqrt(3) - 3/2)
536 // are rationally independent, so no direction of integer components is normal to it.
537 constexpr amrex::Real ref[3] = {1.0_rt, 0.41421356237309515_rt, 0.2320508075688772_rt};
538 amrex::Real m[3] = {0.0_rt, 0.0_rt, 0.0_rt};
539 for (int a = 0; a < n; ++a) {
540 if (!tied(std::abs(h[a][a]), lmax)) {
541 continue;
542 }
543 amrex::Real proj = 0.0_rt;
544 for (int r = 0; r < n; ++r) {
545 proj += ref[r] * v[r][a];
546 }
547 const amrex::Real sgn = (proj >= 0.0_rt) ? 1.0_rt : -1.0_rt;
548 for (int r = 0; r < n; ++r) {
549 m[r] += sgn * v[r][a];
550 }
551 }
552 mx = m[0];
553 my = m[1];
554 mz = m[2];
555}

557/**
558 * \brief Reconstruct the PLIC plane m.x = alpha in cell (i,j,k), in unit-cube cell coordinates with
559 * sum|m| = 1. Reads up to Normal::stencil cells away. Only meaningful in mixed cells
560 * (is_mixed).
561 *
562 * \tparam Normal the normal scheme
563 */
564template <typename Normal>
565AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void fit_plane(const int i,
566 const int j,
567 const int k,
568 amrex::Array4<amrex::Real const> const& F,
569 amrex::Real& mx,
570 amrex::Real& my,
571 amrex::Real& mz,
572 amrex::Real& alpha) {
573 // On a structure one cell thick the normal would be null or noise
574 bool degenerate = thin_structure(i, j, k, F);
575 if (!degenerate) {
576 Normal::normal(i, j, k, F, mx, my, mz);
577 degenerate = std::abs(mx) + std::abs(my) + std::abs(mz) <= std::numeric_limits<amrex::Real>::epsilon();
578 }
579 if (degenerate) {
580 degenerate_normal(i, j, k, F, mx, my, mz);
581 }
582 const amrex::Real norm = std::abs(mx) + std::abs(my) + std::abs(mz);
583 mx /= norm;
584 my /= norm;
585 mz /= norm;
586
587 alpha = volume_intercept(std::abs(mx), std::abs(my), std::abs(mz), F(i, j, k));
588
589 // Back from the reflected (all positive) frame to the cell frame
590 alpha += amrex::min(0.0_rt, mx) + amrex::min(0.0_rt, my) + amrex::min(0.0_rt, mz);
591}

599/**
600 * \brief Call fn with the normal scheme of a reconstruction name, as fn(Normal{}), and return what it
601 * returns. The one place that names the schemes; it aborts on an unknown name. Each path
602 * calls fn once, so it is forwarded there.
603 */
604template <typename Fn>
605decltype(auto) dispatch_normal(std::string const& reconstruction, Fn&& fn) {
606 if (reconstruction == "youngs") {
607 return std::forward<Fn>(fn)(Youngs{});
608 }
609 if (reconstruction == "myc") {
610 return std::forward<Fn>(fn)(MixedYoungsCentered{});
611 }
612 if (reconstruction != "myc_height") {
613 unknown_reconstruction(reconstruction);
614 }
615 return std::forward<Fn>(fn)(HeightFunctionMYC{});
616}
