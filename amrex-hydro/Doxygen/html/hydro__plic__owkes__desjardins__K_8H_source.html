17//-----------------------------------------------------------------------------
18// Unsplit geometric flux kernels for the PLIC volume of fluid, after Owkes &
19// Desjardins (JCP 270, 2014).
20//
21// The liquid crossing a face in a step is the liquid inside the flux volume,
22// the region bounded by the face and its corners traced back along the flow.
23// Everything here works in cell units: cell (i,j,k) is the unit cube
24// [i,i+1] x [j,j+1] x [k,k+1], and face velocities are Courant numbers. In 2D
25// the mesh is one cell thick in z, so that one kernel serves both dimensions.
26//
27// The flux of a face is computed in coordinates relative to the face (see
28// face_liquid_flux), from the displacement of each mesh node over the step.
29// Two boxes sharing a face, or a face and its periodic image, then do the same
30// arithmetic and get the same flux to the bit, so the volume is conserved
31// exactly without synchronizing the faces.
32//
33// Differences from the paper, none of which change the method:
34// - The flux volume is the signed sum of the tetrahedra joining a corner of the
35// face to the triangles of its boundary, rather than the tabulated partition
36// of its Fig. 9. The two are equivalent: any closed oriented triangulated
37// surface gives the right signed integral, including crossed flux volumes.
38// - The side faces are shared by the flux volumes of neighboring faces, so they
39// must be triangulated identically by each of them. The diagonal joins the
40// traced lower node of the mesh edge to its upper node, which depends only
41// on the mesh edge. This is what makes the fluxes tile without overlaps.
42// - Simplices are cut using an explicit stack, for GPU use, and the normals
43// come from the reconstruction in use rather than ELVIRA, as the paper
44// allows.
45//-----------------------------------------------------------------------------
46
47// Tolerances in cell units: positions on a mesh plane (above the round-off in positions of order
48// one), and a traced face with no extent normal to the face.
49#ifdef AMREX_USE_FLOAT
50static constexpr amrex::Real cut_eps = 1.0e-3_rt;
51static constexpr amrex::Real apex_tol = 1.0e-5_rt;
52#else
53static constexpr amrex::Real cut_eps = 1.0e-12_rt;
54static constexpr amrex::Real apex_tol = 1.0e-10_rt;
55#endif

65AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE Vec3 lerp(const Vec3& a,
66 const Vec3& b,
67 const amrex::Real t) noexcept {
68 return Vec3{{a.x[0] + t * (b.x[0] - a.x[0]), a.x[1] + t * (b.x[1] - a.x[1]),
69 a.x[2] + t * (b.x[2] - a.x[2])}};
70}

72/**
73 * \brief Signed volume of the tetrahedron with apex o over the triangle
74 * (a,b,c): positive when the triangle's right-handed normal points away
75 * from o.
76 */
77AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
78signed_volume(const Vec3& o, const Vec3& a, const Vec3& b, const Vec3& c) noexcept {
79 const amrex::Real ax = a.x[0] - o.x[0], ay = a.x[1] - o.x[1], az = a.x[2] - o.x[2];
80 const amrex::Real bx = b.x[0] - o.x[0], by = b.x[1] - o.x[1], bz = b.x[2] - o.x[2];
81 const amrex::Real cx = c.x[0] - o.x[0], cy = c.x[1] - o.x[1], cz = c.x[2] - o.x[2];
82 return (ax * (by * cz - bz * cy) + ay * (bz * cx - bx * cz) + az * (bx * cy - by * cx)) /
83 6.0_rt;
84}

86AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real abs_volume(const Tet& t) noexcept {
87 return std::abs(signed_volume(t.v[0], t.v[1], t.v[2], t.v[3]));
88}

91AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void prism_to_tets(const Vec3& p0,
92 const Vec3& p1,
93 const Vec3& p2,

96 const Vec3& q2,
97 Tet* out) noexcept {
98 out[0] = Tet{{p0, p1, p2, q2}};

103/**
104 * \brief Split a tetrahedron by the plane where the vertex values d vanish.
105 *
106 * The pieces with d <= 0 are written to out first, then those with d > 0.
107 *
108 * \param out room for 6 tetrahedra
109 * \param n_neg,n_pos number of pieces with d <= 0 and with d > 0, at most 3 each
110 */
111AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void
112split_tet(const Tet& t, const amrex::Real* d, Tet* out, int& n_neg, int& n_pos) noexcept {
113 int vin[4];
114 int vout[4];
115 int ni = 0;
116 int no = 0;
117 for (int v = 0; v < 4; ++v) {
118 if (d[v] <= 0.0_rt) {
119 vin[ni++] = v;
120 } else {
121 vout[no++] = v;
122 }
123 }
124 n_neg = 0;
125 n_pos = 0;
126 if (no == 0) {
127 out[0] = t;
128 n_neg = 1;
129 return;
130 }
131 if (ni == 0) {
132 out[0] = t;
133 n_pos = 1;
134 return;
135 }
136
137 // Point where the edge joining an inside and an outside vertex crosses
138 auto cross = [&](const int a, const int b) {
139 return lerp(t.v[a], t.v[b], d[a] / (d[a] - d[b]));
140 };
141 if (ni == 1) {
142 const int a = vin[0];
143 const Vec3 p0 = cross(a, vout[0]);
144 const Vec3 p1 = cross(a, vout[1]);
145 const Vec3 p2 = cross(a, vout[2]);
146 out[0] = Tet{{t.v[a], p0, p1, p2}};
147 n_neg = 1;
148 prism_to_tets(t.v[vout[0]], t.v[vout[1]], t.v[vout[2]], p0, p1, p2, out + 1);
149 n_pos = 3;
150 } else if (no == 1) {
151 const int a = vout[0];
152 const Vec3 p0 = cross(vin[0], a);
153 const Vec3 p1 = cross(vin[1], a);
154 const Vec3 p2 = cross(vin[2], a);
155 prism_to_tets(t.v[vin[0]], t.v[vin[1]], t.v[vin[2]], p0, p1, p2, out);
156 n_neg = 3;
157 out[3] = Tet{{t.v[a], p0, p1, p2}};
158 n_pos = 1;
159 } else {
160 const int a = vin[0];
161 const int b = vin[1];
162 const int c = vout[0];
163 const int e = vout[1];
164 const Vec3 ac = cross(a, c);
165 const Vec3 ae = cross(a, e);
166 const Vec3 bc = cross(b, c);
167 const Vec3 be = cross(b, e);
168 prism_to_tets(t.v[a], ac, ae, t.v[b], bc, be, out);
169 n_neg = 3;
170 prism_to_tets(t.v[c], ac, bc, t.v[e], ae, be, out + 3);
171 n_pos = 3;
172 }
173}

175/**
176 * \brief Volume of the part of a tetrahedron of volume vol on the liquid side of the plane
177 * m.(x - x_lo) <= alpha of the cell with low corner x_lo.
178 *
179 * With one vertex on the liquid side, that part is the corner of the tetrahedron at the vertex,
180 * cut at a fraction r of each edge from it: vol times the product of the r. With one vertex on the
181 * gas side, it is the rest of the tetrahedron. With two on each side, it is the prism of split_tet
182 * on the liquid side.
183 */
184AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real tet_liquid_volume(const Tet& t,
185 const amrex::Real vol,
186 const int i,
187 const int j,
188 const int k,
189 const amrex::Real* plane) noexcept {
190 amrex::Real d[4];
191 int vin[4];
192 int vout[4];
193 int ni = 0;
194 int no = 0;
195 for (int v = 0; v < 4; ++v) {
196 d[v] = plane[0] * (t.v[v].x[0] - static_cast<amrex::Real>(i)) +
197 plane[1] * (t.v[v].x[1] - static_cast<amrex::Real>(j)) +
198 plane[2] * (t.v[v].x[2] - static_cast<amrex::Real>(k)) - plane[3];
199 if (d[v] <= 0.0_rt) {
200 vin[ni++] = v;
201 } else {
202 vout[no++] = v;
203 }
204 }
205 if (no == 0) {
206 return vol;
207 }
208 if (ni == 0) {
209 return 0.0_rt;
210 }
211 // Fraction of the edge from vertex a, on one side, to vertex b, on the other, before the plane.
212 // d[a] and d[b] have opposite signs, so the fraction is in [0,1].
213 auto frac = [&](const int a, const int b) { return d[a] / (d[a] - d[b]); };
214 if (ni == 1) {
215 const int a = vin[0];
216 return vol * (frac(a, vout[0]) * frac(a, vout[1]) * frac(a, vout[2]));
217 }
218 if (no == 1) {
219 const int b = vout[0];
220 return vol * (1.0_rt - frac(b, vin[0]) * frac(b, vin[1]) * frac(b, vin[2]));
221 }
222 const int a = vin[0];
223 const int b = vin[1];
224 const int c = vout[0];
225 const int e = vout[1];
226 Tet prism[3];
227 prism_to_tets(t.v[a], lerp(t.v[a], t.v[c], frac(a, c)), lerp(t.v[a], t.v[e], frac(a, e)), t.v[b],
228 lerp(t.v[b], t.v[c], frac(b, c)), lerp(t.v[b], t.v[e], frac(b, e)), prism);
229 return abs_volume(prism[0]) + abs_volume(prism[1]) + abs_volume(prism[2]);
230}

232/** \brief Liquid volume of a piece of tetrahedron lying within cell (i,j,k). */
233AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
234piece_liquid(const Tet& t,
235 const amrex::Real vol,
236 const int i,
237 const int j,
238 const int k,
239 amrex::Array4<amrex::Real const> const& F,
240 amrex::Array4<amrex::Real const> const& plane) noexcept {
241 const amrex::Real f = F(i, j, k);
242 if (f <= pure_tol) {
243 return 0.0_rt;
244 }
245 if (f >= 1.0_rt - pure_tol) {
246 return vol;
247 }
248 const amrex::Real pl[4] = {plane(i, j, k, 0), plane(i, j, k, 1), plane(i, j, k, 2),
249 plane(i, j, k, 3)};
250 return tet_liquid_volume(t, vol, i, j, k, pl);
251}

253// Tetrahedra held while cutting by the mesh planes. Each cut replaces one
254// tetrahedron by at most 6 and a depth-first path crosses each mesh plane at
255// most once. For a Courant number below 1 a flux volume lies within the cells
256// around its face (no node moves a cell, and the correction apex is held there,
257// see flux_volume): it crosses at most 1 mesh plane along the face normal
258// and 2 along each other direction, 5 in all, and the stack holds at most
259// 1 + 5 * 5 = 26 tetrahedra. Kept small, as each GPU thread holds its own.
260static constexpr int cut_stack_size = 32;
261
262/**
263 * \brief Liquid volume of a tetrahedron: cut along the mesh planes into one
264 * piece per cell, each with the liquid of its cell (piece_liquid).
265 *
266 * \param bx cells where F and plane are valid; pieces beyond it are charged to
267 * the nearest cell inside it
268 */
269AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
270cut_tet_liquid(const Tet& tet,
271 const amrex::Box& bx,
272 amrex::Array4<amrex::Real const> const& F,
273 amrex::Array4<amrex::Real const> const& plane) noexcept {
274 amrex::Real liquid = 0.0_rt;
275 Tet stack[cut_stack_size];
276 int top = 0;
277 stack[top++] = tet;
278 const auto lo = amrex::lbound(bx);
279 const auto hi = amrex::ubound(bx);
280
281 while (top > 0) {
282 const Tet t = stack[--top];
283
284 // The first mesh plane crossing this tetrahedron, if any. The z planes
285 // play no part in 2D, whose mesh is one cell thick.
286 int cut_dir = -1;
287 amrex::Real cut_at = 0.0_rt;
288 for (int d = 0; d < AMREX_SPACEDIM && cut_dir < 0; ++d) {
289 amrex::Real tmin = t.v[0].x[d];
290 amrex::Real tmax = t.v[0].x[d];
291 for (int v = 1; v < 4; ++v) {
292 tmin = amrex::min(tmin, t.v[v].x[d]);
293 tmax = amrex::max(tmax, t.v[v].x[d]);
294 }
295 const amrex::Real p = std::floor(tmin + cut_eps) + 1.0_rt;
296 if (tmax > p + cut_eps) {
297 cut_dir = d;
298 cut_at = p;
299 }
300 }
301
302 // A full stack means the flux volume reaches beyond the cells around its face, which the
303 // check on the Courant numbers rules out in every build. Should it happen anyway, a release
304 // build charges the uncut piece to the cell of its centroid.
305 AMREX_ASSERT_WITH_MESSAGE(cut_dir < 0 || top + 6 <= cut_stack_size,
306 "PLIC: flux volume needs more cuts than the stack holds");
307 if (cut_dir >= 0 && top + 6 <= cut_stack_size) {
308 const amrex::Real d[4] = {t.v[0].x[cut_dir] - cut_at, t.v[1].x[cut_dir] - cut_at,
309 t.v[2].x[cut_dir] - cut_at, t.v[3].x[cut_dir] - cut_at};
310 int n_neg = 0;
311 int n_pos = 0;
312 split_tet(t, d, stack + top, n_neg, n_pos);
313 top += n_neg + n_pos;
314 continue;
315 }
316
317 // Within a single cell, found from the centroid
318 const amrex::Real vol = abs_volume(t);
319 if (vol <= 0.0_rt) {
320 continue;
321 }
322 int c[3] = {0, 0, 0};
323 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
324 c[d] = static_cast<int>(
325 std::floor(0.25_rt * (t.v[0].x[d] + t.v[1].x[d] + t.v[2].x[d] + t.v[3].x[d])));
326 }
327 const int ci = amrex::Clamp(c[0], lo.x, hi.x);
328 const int cj = amrex::Clamp(c[1], lo.y, hi.y);
329 const int ck = amrex::Clamp(c[2], lo.z, hi.z);
330 liquid += piece_liquid(t, vol, ci, cj, ck, F, plane);
331 }
332 return liquid;
333}

335/**
336 * \brief Apex o of the two tetrahedra (o,a,c,e) and (o,a,b,c) added on a
337 * traced face (a,b,c,e) to give them a total signed volume vcor. o lies
338 * over the barycenter of the face; the volume is linear in o along dir.
339 *
340 * \param tri the triangles (a,c,e) and (a,b,c) of the traced face, with +dir normals
341 * \return false if the face has no extent normal to dir (no correction)
342 */
343AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool correction_apex(const Vec3* const (&tri)[2][3],
344 const int dir,
345 const amrex::Real vcor,
346 Vec3& o) noexcept {
347 const Vec3& a = *tri[0][0];
348 const Vec3& b = *tri[1][1];
349 const Vec3& c = *tri[0][1];
350 const Vec3& e = *tri[0][2];
351 for (int d = 0; d < 3; ++d) {
352 o.x[d] = 0.25_rt * (a.x[d] + b.x[d] + c.x[d] + e.x[d]);
353 }
354 amrex::Real s0 = 0.0_rt; // 6 V at o_dir = 0
355 amrex::Real sd = 0.0_rt; // d(6 V)/d(o_dir)
356 for (const auto& n : tri) {
357 const Vec3 &p = *n[0], &q = *n[1], &r = *n[2];
358 const amrex::Real u[3] = {q.x[0] - p.x[0], q.x[1] - p.x[1], q.x[2] - p.x[2]};
359 const amrex::Real w[3] = {r.x[0] - p.x[0], r.x[1] - p.x[1], r.x[2] - p.x[2]};
360 const amrex::Real nrm[3] = {u[1] * w[2] - u[2] * w[1], u[2] * w[0] - u[0] * w[2],
361 u[0] * w[1] - u[1] * w[0]};
362 // V = (p - o).N / 6
363 for (int d = 0; d < 3; ++d) {
364 const amrex::Real od = (d == dir) ? 0.0_rt : o.x[d];
365 s0 += (p.x[d] - od) * nrm[d];
366 }
367 sd -= nrm[dir];
368 }
369 if (std::abs(sd) <= apex_tol) {
370 return false;
371 }
372 o.x[dir] = (6.0_rt * vcor - s0) / sd;
373 return true;
374}

376// Tetrahedra of a flux volume: the fan and the two of the correction
377static constexpr int max_flux_tets = 10;
378
379/**
380 * \brief A flux volume as signed tetrahedra over 9 points: the face corners a, b, c, e
381 * counterclockwise about +dir (points 0-3), their traced positions a', b', c', e' (4-7),
382 * and the apex o of the correction (8).
383 */
384struct FluxVolume {
385 Vec3 p[9];
386 //! Signed volume of each tetrahedron
387 amrex::Real vol[max_flux_tets];
388 //! Number of tetrahedra: the 8 of the fan, and the 2 of the correction if there is one
389 int ntet;
390 //! Whether the volume of the tetrahedra is the Courant number: it falls short where the apex
391 //! is held near the face, or there is no correction
392 bool matches_courant;
393
394 //! Number of points in use: o only with a correction
395 [[nodiscard]] AMREX_GPU_HOST_DEVICE int npoint() const noexcept { return (ntet > 8) ? 9 : 8; }
396
397 /**
398 * \brief Tetrahedron n: the fan of the boundary from a, with outward triangles, then the
399 * correction from o over the traced face. Triangles touching a vanish: the face
400 * itself and the side faces through a. The diagonals of the side faces depend only on
401 * the mesh edge (see above).
402 */
403 [[nodiscard]] AMREX_GPU_HOST_DEVICE Tet tet(const int n) const noexcept {
404 constexpr int v[max_flux_tets][4] = {{0, 4, 5, 1}, // side a-b, diagonal a' - b
405 {0, 3, 7, 4}, // side e-a, diagonal e - a'
406 {0, 1, 5, 2}, // side b-c, diagonal b' - c
407 {0, 5, 6, 2}, //
408 {0, 2, 6, 7}, // side c-e, diagonal e' - c
409 {0, 2, 7, 3}, //
410 {0, 4, 7, 6}, // traced face, diagonal a' - c'
411 {0, 4, 6, 5}, //
412 {8, 4, 6, 7}, // correction, +dir normals
413 {8, 4, 5, 6}};
414 return Tet{{p[v[n][0]], p[v[n][1]], p[v[n][2]], p[v[n][3]]}};
415 }
416};

418/**
419 * \brief The flux volume of the low face of cell (i,j,k) in direction dir.
420 *
421 * \param disp displacement of each mesh node over the step, from its traced
422 * position at the start of the step, in cell units (3 components)
423 * \param courant the face velocity times dt/dx, which the flux volume is
424 * corrected to match
425 */
426AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void
427flux_volume(const int i,
428 const int j,
429 const int k,
430 const int dir,
431 const amrex::Real courant,
432 amrex::Array4<amrex::Real const> const& disp,
433 FluxVolume& fv) noexcept {
434 // Right-handed tangential directions. The face spans 0..1 along each, and
435 // in 2D the node arrays have no extent in z: the k+1 nodes are the k nodes
436 // shifted by one cell.
437 const int t1 = (dir + 1) % 3;
438 const int t2 = (dir + 2) % 3;
439
440 int off1[3] = {0, 0, 0};
441 int off2[3] = {0, 0, 0};
442 off1[t1] = 1;
443 off2[t2] = 1;
444
445 auto node = [&](const int s1, const int s2) {
446 return Vec3{{static_cast<amrex::Real>(i + s1 * off1[0] + s2 * off2[0]),
447 static_cast<amrex::Real>(j + s1 * off1[1] + s2 * off2[1]),
448 static_cast<amrex::Real>(k + s1 * off1[2] + s2 * off2[2])}};
449 };
450 auto traced = [&](const int s1, const int s2) {
451 const Vec3 p = node(s1, s2);
452 const int ni = i + s1 * off1[0] + s2 * off2[0];
453 const int nj = j + s1 * off1[1] + s2 * off2[1];
454 // Nodes of a 2D mesh: z stays as it was, and only the in-plane offset selects the node
455 constexpr bool is3d = (AMREX_SPACEDIM == 3);
456 const int nk = is3d ? k + s1 * off1[2] + s2 * off2[2] : 0;
457 return Vec3{{p.x[0] + disp(ni, nj, nk, 0), p.x[1] + disp(ni, nj, nk, 1),
458 is3d ? p.x[2] + disp(ni, nj, nk, 2) : p.x[2]}};
459 };
460
461 // Face corners counterclockwise about +dir, and their traced positions
462 Vec3* const p = fv.p;
463 p[0] = node(0, 0);
464 p[1] = node(1, 0);
465 p[2] = node(1, 1);
466 p[3] = node(0, 1);
467 p[4] = traced(0, 0);
468 p[5] = traced(1, 0);
469 p[6] = traced(1, 1);
470 p[7] = traced(0, 1);
471
472 // Fan of the boundary from a. With outward triangles the signed volumes
473 // add up to the flux volume, positive for a flow along +dir.
474 amrex::Real vol = 0.0_rt;
475 for (int n = 0; n < 8; ++n) {
476 const Tet t = fv.tet(n);
477 fv.vol[n] = signed_volume(t.v[0], t.v[1], t.v[2], t.v[3]);
478 vol += fv.vol[n];
479 }
480
481 // Extra volume so that the flux volume matches the face velocity exactly:
482 // two tetrahedra from the traced face to a point o above its barycenter, in
483 // the direction of dir (Eq. 27-28). Their volume is linear in o along dir.
484 const Vec3* const ttri[2][3] = {{&p[4], &p[6], &p[7]}, {&p[4], &p[5], &p[6]}}; // +dir normals
485 Vec3& o = p[8];
486 const bool has_cor = correction_apex(ttri, dir, courant - vol, o);
487 // The rest of the flux volume lies within a cell of the face, as no node moves a cell. Where the
488 // traced face has almost no extent normal to dir, the apex would lie far beyond, so it is held
489 // there too: the flux volume then falls short of the Courant number (see flux_volume_liquid), but
490 // every face still reads only the cells around it, and the boxes sharing a face clamp it the same
491 // way.
492 const amrex::Real o_dir = o.x[dir];
493 o.x[dir] = amrex::Clamp(o.x[dir], p[0].x[dir] - 1.0_rt, p[0].x[dir] + 1.0_rt);
494 fv.matches_courant = has_cor && o.x[dir] == o_dir;
495
496 fv.ntet = has_cor ? max_flux_tets : 8;
497 for (int n = 8; n < fv.ntet; ++n) {
498 const Tet t = fv.tet(n);
499 fv.vol[n] = signed_volume(t.v[0], t.v[1], t.v[2], t.v[3]);
500 }
501}

503/**
504 * \brief Whether the cells [lo, hi], limited to bx, are all gas or all liquid.
505 *
506 * \return true if they are pure, with a_liquid set to 1 or 0 accordingly
507 */
508AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool pure_cells(const int* lo,
509 const int* hi,
510 amrex::Array4<amrex::Real const> const& F,
511 const amrex::Box& bx,
512 bool& a_liquid) noexcept {
513 const auto blo = amrex::lbound(bx);
514 const auto bhi = amrex::ubound(bx);
515 bool all_gas = true;
516 bool all_liq = true;
517 // In 2D bx is one cell thick, at k = 0
518 const int klo = amrex::Clamp(lo[2], blo.z, bhi.z);
519 const int khi = amrex::Clamp(hi[2], blo.z, bhi.z);
520 const int ilo = amrex::Clamp(lo[0], blo.x, bhi.x);
521 const int ihi = amrex::Clamp(hi[0], blo.x, bhi.x);
522 const int jlo = amrex::Clamp(lo[1], blo.y, bhi.y);
523 const int jhi = amrex::Clamp(hi[1], blo.y, bhi.y);
524 for (int kk = klo; kk <= khi; ++kk) {
525 for (int jj = jlo; jj <= jhi; ++jj) {
526 for (int ii = ilo; ii <= ihi; ++ii) {
527 const amrex::Real f = F(ii, jj, kk);
528 all_gas = all_gas && (f <= pure_tol);
529 all_liq = all_liq && (f >= 1.0_rt - pure_tol);
530 }
531 if (!all_gas && !all_liq) {
532 return false;
533 }
534 }
535 }
536 a_liquid = all_liq;
537 return true;
538}

540/**
541 * \brief Whether a flux volume lies in pure cells only, from its bounding box:
542 * it then needs no cutting.
543 *
544 * \param liquid its signed liquid volume, if so
545 */
546AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool
547pure_flux_volume(const FluxVolume& fv,
548 const amrex::Real courant,
549 amrex::Array4<amrex::Real const> const& F,
550 const amrex::Box& bx,
551 amrex::Real& liquid) noexcept {
552 amrex::Real pmin[3] = {fv.p[0].x[0], fv.p[0].x[1], fv.p[0].x[2]};
553 amrex::Real pmax[3] = {pmin[0], pmin[1], pmin[2]};
554 for (int n = 1; n < fv.npoint(); ++n) {
555 for (int d = 0; d < 3; ++d) {
556 pmin[d] = amrex::min(pmin[d], fv.p[n].x[d]);
557 pmax[d] = amrex::max(pmax[d], fv.p[n].x[d]);
558 }
559 }
560 // The cells the bounding box overlaps: a vertex on a mesh plane does not bring in the cell
561 // beyond it, as the downwind cell of a flux volume that ends on its face
562 int lo[3];
563 int hi[3];
564 for (int d = 0; d < 3; ++d) {
565 lo[d] = static_cast<int>(std::floor(pmin[d]));
566 hi[d] = amrex::max(lo[d], static_cast<int>(std::ceil(pmax[d])) - 1);
567 }
568 bool is_liquid = false;
569 if (!pure_cells(lo, hi, F, bx, is_liquid)) {
570 return false;
571 }
572 liquid = is_liquid ? courant : 0.0_rt;
573 return true;
574}

576/**
577 * \brief F of the upwind cell of the low face of cell iv in direction dir, for a flow u along dir:
578 * the cell below the face for u >= 0. It is held to [0,1], as an edge state is, though F may
579 * be off it by round-off.
580 */
581AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
582upwind_fraction(amrex::Array4<amrex::Real const> const& F,
583 amrex::IntVect iv,
584 const int dir,
585 const amrex::Real u) noexcept {
586 if (u >= 0.0_rt) {
587 iv[dir] -= 1;
588 }
589 return amrex::Clamp(F(iv), 0.0_rt, 1.0_rt);
590}

592/**
593 * \brief Liquid volume in the flux volume of the low face of cell (i,j,k) in direction dir, per
594 * cell volume.
595 *
596 * Where the flux volume falls short of the Courant number (see flux_volume), the liquid is its
597 * liquid fraction times the Courant number, so that a face in liquid only carries the Courant number,
598 * as on the faces that need no cuts: a cell full of liquid stays full. Without a volume to take a
599 * fraction of, the fraction is F of the upwind cell.
600 *
601 * \param bx cells where F and plane are valid
602 * \return signed liquid volume crossing the face
603 */
604AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::Real
605flux_volume_liquid(const int i,
606 const int j,
607 const int k,
608 const int dir,
609 const amrex::Real courant,
610 amrex::Array4<amrex::Real const> const& F,
611 amrex::Array4<amrex::Real const> const& plane,
612 amrex::Array4<amrex::Real const> const& disp,
613 const amrex::Box& bx) noexcept {
614 FluxVolume fv;
615 flux_volume(i, j, k, dir, courant, disp, fv);
616 amrex::Real liquid = 0.0_rt;
617 // All in one phase: no cuts needed
618 if (pure_flux_volume(fv, courant, F, bx, liquid)) {
619 return liquid;
620 }
621 amrex::Real volume = 0.0_rt;
622 for (int n = 0; n < fv.ntet; ++n) {
623 const amrex::Real v = fv.vol[n];
624 volume += v;
625 if (v == 0.0_rt) {
626 continue;
627 }
628 const amrex::Real s = (v > 0.0_rt) ? 1.0_rt : -1.0_rt;
629 liquid += s * cut_tet_liquid(fv.tet(n), bx, F, plane);
630 }
631 if (!fv.matches_courant) {
632 amrex::Real fraction = 0.0_rt;
633 if (std::abs(volume) > apex_tol) {
634 fraction = liquid / volume;
635 } else {
636 fraction = upwind_fraction(F, amrex::IntVect(AMREX_D_DECL(i, j, k)), dir, courant);
637 }
638 liquid = courant * amrex::Clamp(fraction, 0.0_rt, 1.0_rt);
639 }
640 return liquid;
641}

679/**
680 * \brief Displacement of mesh node (i,j,k) over the step, in cell units, by the midpoint rule: the
681 * velocity at the node, then the velocity half a step back along it. Velocities between
682 * faces are multilinear interpolations of the face velocities themselves, so that a node of
683 * a box reads no faces beyond the first ghost layer, and the arithmetic is relative to the
684 * node, so that every box holding it, and its periodic images, get the same displacement.
685 *
686 * \param umac face velocities in each direction
687 * \param dtdx dt / dx in each direction, to Courant numbers
688 * \param faces faces normal to each direction that may be read: a face beyond them, across a
689 * wall, is replaced by the nearest face on the domain boundary. The node reads the
690 * faces up to one beyond it in every direction.
691 * \param disp output, 3 components (the third is 0 in 2D)
692 */
693AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void
694node_displacement(const int i,
695 const int j,
696 const int k,
697 amrex::GpuArray<amrex::Array4<amrex::Real const>, AMREX_SPACEDIM> const& umac,
698 amrex::GpuArray<amrex::Real, AMREX_SPACEDIM> const& dtdx,
699 amrex::GpuArray<amrex::Box, AMREX_SPACEDIM> const& faces,
700 amrex::Real* disp) noexcept {
701 const int node[3] = {i, j, k};
702 // The Courant number of the face normal to d at node + off (normal) and cell node + off
703 // (tangential)
704 auto face = [&](const int d, const int* off) {
705 int f[3] = {node[0] + off[0], node[1] + off[1], node[2] + off[2]};
706 for (int t = 0; t < AMREX_SPACEDIM; ++t) {
707 f[t] = amrex::Clamp(f[t], faces[d].smallEnd(t), faces[d].bigEnd(t));
708 }
709 // In 2D, k and off[2] are 0
710 return umac[d](f[0], f[1], f[2]) * dtdx[d];
711 };
712 // Velocity at x, relative to the node, |x| < 1/2 in each direction. Face centers normal to d lie
713 // at integer positions along d and half-integer positions along the others.
714 auto velocity = [&](const amrex::Real* x, amrex::Real* u) {
715 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
716 const int b = (x[d] < 0.0_rt) ? -1 : 0;
717 const amrex::Real w[3] = {
718 (d == 0) ? x[0] - static_cast<amrex::Real>(b) : x[0] + 0.5_rt,
719 (d == 1) ? x[1] - static_cast<amrex::Real>(b) : x[1] + 0.5_rt,
720 (d == 2) ? x[2] - static_cast<amrex::Real>(b) : x[2] + 0.5_rt};
721 const int base[3] = {(d == 0) ? b : -1, (d == 1) ? b : -1, (d == 2) ? b : -1};
722 amrex::Real sum = 0.0_rt;
723#if AMREX_SPACEDIM == 2
724 for (int s1 = 0; s1 <= 1; ++s1) {
725 for (int s0 = 0; s0 <= 1; ++s0) {
726 const int off[3] = {base[0] + s0, base[1] + s1, 0};
727 sum += (s0 != 0 ? w[0] : 1.0_rt - w[0]) * (s1 != 0 ? w[1] : 1.0_rt - w[1]) *
728 face(d, off);
729 }
730 }
731#else
732 for (int s2 = 0; s2 <= 1; ++s2) {
733 for (int s1 = 0; s1 <= 1; ++s1) {
734 for (int s0 = 0; s0 <= 1; ++s0) {
735 const int off[3] = {base[0] + s0, base[1] + s1, base[2] + s2};
736 sum += (s0 != 0 ? w[0] : 1.0_rt - w[0]) * (s1 != 0 ? w[1] : 1.0_rt - w[1]) *
737 (s2 != 0 ? w[2] : 1.0_rt - w[2]) * face(d, off);
738 }
739 }
740 }
741#endif
742 u[d] = sum;
743 }
744 };
745 const amrex::Real x0[3] = {0.0_rt, 0.0_rt, 0.0_rt};
746 amrex::Real u0[3] = {0.0_rt, 0.0_rt, 0.0_rt};
747 velocity(x0, u0);
748 const amrex::Real xm[3] = {-0.5_rt * u0[0], -0.5_rt * u0[1], -0.5_rt * u0[2]};
749 amrex::Real um[3] = {0.0_rt, 0.0_rt, 0.0_rt};
750 velocity(xm, um);
751 for (int d = 0; d < 3; ++d) {
752 disp[d] = -um[d];
753 }
754}
